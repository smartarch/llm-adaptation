# Strategy reasoning (embedded as comments for clarity)
# - We now allocate drones to protect fields in order of decreasing threat level.
# - Drones already protecting a field stay assigned to that field (to maintain coverage).
# - We first ensure the most threatened field reaches full protection if possible.
# - Then we allocate any remaining drones to the next most-threatened fields in order,
#   again trying to reach full protection for each until drones run out.
# - Any drones not allocated to any protected field are set to idle.
# - This avoids leaving drones idle when there are still protectable fields with positive threat.

from typing import List

from generated_adaptations.base_classes.farm import FarmAdaptation  # type: ignore

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components: List, environment, group_ids: List[str], step: int):
        # Gather fields with positive threat, sorted by threat level (descending)
        fields_with_threat = [f for f in environment.fields if getattr(f, "threat_level", 0.0) > 0.0]
        fields_with_threat.sort(key=lambda f: float(getattr(f, "threat_level", 0.0)), reverse=True)

        if not fields_with_threat:
            # No field needs protection; idle all drones
            for c in components:
                environment.assign_group(c, "idle")
            return

        # Helper: count how many drones are currently protecting a given field
        def currently_protecting(field_id: str) -> int:
            return sum(
                1 for c in components
                if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) == field_id
            )

        # First, ensure drones already protecting stay in place
        for c in components:
            if getattr(c, "state", None) == "protecting":
                # If this drone is protecting a field, keep it in that field's group
                field_id = getattr(c, "target_id", None)
                if field_id is not None:
                    environment.assign_group(c, f"protecting {field_id}")
                else:
                    environment.assign_group(c, "idle")
            else:
                # Non-protecting drones will be reassigned in the allocation passes below
                pass

        # Total drones available for allocation (excluding those already protecting)
        # We will re-use the same components list in a two-pass approach:
        # Pass 1: note which drones are already protecting; Pass 2: allocate the rest.
        protecting_by_id = {
            getattr(c, "target_id", None): c
            for c in components
            if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) is not None
        }

        # Mark drones already protecting as assigned; collect others for redistribution
        unassigned_drones = []
        for c in components:
            if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) is not None:
                # Already assigned earlier in this pass
                environment.assign_group(c, f"protecting {getattr(c, 'target_id')}")  # ensure group is set
            else:
                unassigned_drones.append(c)

        # Allocate remaining drones to fields in threat order
        for field in fields_with_threat:
            field_id = field.id
            current = currently_protecting(field_id)
            required = getattr(field, "drones_for_full_protection", 0)
            try:
                required = int(required)
            except Exception:
                required = 0

            needed = max(0, required - current)
            if needed == 0:
                continue

            # Give drones to this field until full or we run out
            i = 0
            while needed > 0 and i < len(unassigned_drones):
                drone = unassigned_drones[i]
                environment.assign_group(drone, f"protecting {field_id}")
                # Move this drone from unassigned to assigned; remove from list by popping
                unassigned_drones.pop(i)
                # This drone is now assigned; do not increment i since list shifted
                needed -= 1

        # Any remaining drones (unassigned) become idle
        for drone in unassigned_drones:
            environment.assign_group(drone, "idle")
