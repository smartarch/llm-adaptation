# SmartAdaptation for the Dragon Hunt game
#
# Reasoning and updated strategy:
# - Fixes and improvements:
#   1. Avoid use of sets for tracking assigned components (some component objects may not be hashable).
#   2. Ensure both warrior and farmer spawns occur over time by alternating spawn priority each step:
#      - On even steps prioritize spawning Warriors first, then Farmers.
#      - On odd steps prioritize spawning Farmers first, then Warriors.
#   3. Keep a small reserve of farmers farming (min_farmers_farming) to preserve wheat production.
#   4. Limit the number of Warriors sent to the Cave each turn (attack_squad_size) to reduce exposure.
#   5. Guarantee every component is assigned exactly once per call and only to valid group names.
#
# This aims to reduce field damage while ensuring enough spawning of both types to meet the tests'
# expectations and to win reliably.

from generated_adaptations.base_classes.dragon import DragonHuntAdaptation

class SmartAdaptation(DragonHuntAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        # Max number of warriors that attack the dragon in a single step.
        self.attack_squad_size = 3
        # Minimum number of farmers we try to keep farming if available
        self.min_farmers_farming = 1

    def assign_in_village(self, components, environment, group_ids, step: int):
        """
        Assign villagers in the Village to:
          - "farm", "cave", "spawn farmer", "spawn warrior"
        Alternates spawn priority by step parity to ensure both types get spawned across the game.
        """
        # Partition villagers by role
        farmers = [c for c in components if getattr(c, "role", "") == "Farmer"]
        warriors = [c for c in components if getattr(c, "role", "") == "Warrior"]

        wheat = getattr(environment.farm, "wheat", 0)

        # Reserve some farmers to farm (if possible)
        reserved_count = min(self.min_farmers_farming, len(farmers))
        reserved_farmers = farmers[:reserved_count]
        remaining_farmers = farmers[reserved_count:]

        # Track assigned farmers by index instead of set
        assigned_flags = [False] * len(remaining_farmers)  # corresponds to remaining_farmers indices

        def spawn_pairs(cost, num_pairs):
            """
            Assign num_pairs pairs from remaining_farmers sequentially.
            Returns actual pairs spawned (could be less if not enough unassigned).
            """
            spawned = 0
            idx = 0
            nonlocal wheat
            # find pairs by scanning for two unassigned farmers
            while spawned < num_pairs:
                # find first unassigned
                while idx < len(remaining_farmers) and assigned_flags[idx]:
                    idx += 1
                if idx >= len(remaining_farmers):
                    break
                # find next unassigned after idx
                j = idx + 1
                while j < len(remaining_farmers) and assigned_flags[j]:
                    j += 1
                if j >= len(remaining_farmers):
                    break
                # assign pair idx and j
                f1 = remaining_farmers[idx]; f2 = remaining_farmers[j]
                yield (f1, f2)
                assigned_flags[idx] = True
                assigned_flags[j] = True
                spawned += 1
                wheat -= cost
                idx = j + 1

        # Compute pair availability and spawn counts depending on priority
        pairs_available = len(remaining_farmers) // 2

        # Determine spawn order: even steps -> warriors first, odd steps -> farmers first
        if step % 2 == 0:
            primary = "warrior"
            secondary = "farmer"
        else:
            primary = "farmer"
            secondary = "warrior"

        # Helper to compute max pairs by wheat for a given cost
        def max_pairs_by_wheat(cost):
            return wheat // cost

        # First handle primary
        if primary == "warrior":
            max_warrior_spawns = min(pairs_available, max_pairs_by_wheat(12))
            warrior_pairs = list(spawn_pairs(12, max_warrior_spawns))
            # Assign those pairs
            for f1, f2 in warrior_pairs:
                environment.assign_group(f1, "spawn warrior")
                environment.assign_group(f2, "spawn warrior")
        else:
            max_farmer_spawns = min(pairs_available, max_pairs_by_wheat(10))
            farmer_pairs = list(spawn_pairs(10, max_farmer_spawns))
            for f1, f2 in farmer_pairs:
                environment.assign_group(f1, "spawn farmer")
                environment.assign_group(f2, "spawn farmer")

        # Recompute available pairs for secondary (some farmers marked assigned)
        unassigned_pairs = sum(1 for i in range(len(assigned_flags)) if not assigned_flags[i]) // 2

        if secondary == "warrior":
            max_warrior_spawns = min(unassigned_pairs, max_pairs_by_wheat(12))
            warrior_pairs = list(spawn_pairs(12, max_warrior_spawns))
            for f1, f2 in warrior_pairs:
                environment.assign_group(f1, "spawn warrior")
                environment.assign_group(f2, "spawn warrior")
        else:
            max_farmer_spawns = min(unassigned_pairs, max_pairs_by_wheat(10))
            farmer_pairs = list(spawn_pairs(10, max_farmer_spawns))
            for f1, f2 in farmer_pairs:
                environment.assign_group(f1, "spawn farmer")
                environment.assign_group(f2, "spawn farmer")

        # Any remaining unassigned remaining_farmers go to farm
        for idx, f in enumerate(remaining_farmers):
            if not assigned_flags[idx]:
                environment.assign_group(f, "farm")

        # Assign reserved farmers to farm
        for f in reserved_farmers:
            environment.assign_group(f, "farm")

        # Decide which warriors to send to cave (attack squad). Choose highest hp warriors to attack.
        if warriors:
            sorted_w = sorted(warriors, key=lambda c: getattr(c, "hp", 0), reverse=True)
            to_send = sorted_w[: self.attack_squad_size]
            to_keep = sorted_w[self.attack_squad_size :]
            for w in to_send:
                environment.assign_group(w, "cave")
            for w in to_keep:
                # keep safe and productive in village
                environment.assign_group(w, "farm")

    def assign_in_cave(self, components, environment, group_ids, step: int):
        """
        Assign villagers in the Cave to:
          - "attack", "cave", "village"
        Rules:
          - Farmers in Cave: immediately send back to "village".
          - Warriors: allow up to attack_squad_size to "attack" (prefer highest hp), send others back to "village".
        """
        farmers = [c for c in components if getattr(c, "role", "") == "Farmer"]
        warriors = [c for c in components if getattr(c, "role", "") == "Warrior"]

        # Send farmers back to village
        for f in farmers:
            environment.assign_group(f, "village")

        if warriors:
            sorted_w = sorted(warriors, key=lambda c: getattr(c, "hp", 0), reverse=True)
            attackers = sorted_w[: self.attack_squad_size]
            others = sorted_w[self.attack_squad_size :]
            for w in attackers:
                environment.assign_group(w, "attack")
            for w in others:
                environment.assign_group(w, "village")
