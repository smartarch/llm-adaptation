"""
Adaptive strategy (robust top-field protection with deterministic ordering):
- Determine the most threatened field (tie-break deterministically by numeric part of the field id).
- Fully protect that top field whenever possible by reassigning drones:
  - Preserve drones already protecting the top field.
  - If more drones are needed to reach full protection, allocate the closest available drones to the top field.
- If there are not enough drones to reach full protection, protect as many as possible.
- All drones not allocated to the top field become idle.
- This approach prioritizes the most dangerous field and minimizes drone movement while avoiding illegal group assignments.
"""

import math
import re
from generated_adaptations.base_classes import farm as base_farm

class SmartFarmAdaptation(base_farm.FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def _field_numeric_id(self, field_id):
        # Try to extract trailing digits from an id like "Field_1" or "Field10"
        if isinstance(field_id, str):
            m = re.search(r'(\d+)$', field_id)
            if m:
                try:
                    return int(m.group(1))
                except ValueError:
                    pass
        return 0

    def assign_drones(self, components, environment, group_ids, step: int):
        # Gather fields with positive threat
        fields_with_threat = [f for f in environment.fields if getattr(f, "threat_level", 0) > 0]

        # If no field is threatened, idle all drones
        if not fields_with_threat:
            for c in components:
                environment.assign_group(c, "idle")
            return

        # Pick the most threatened field (stable deterministic tie-break by numeric id)
        top_field = max(
            fields_with_threat,
            key=lambda f: (getattr(f, "threat_level", 0), -self._field_numeric_id(getattr(f, "id", "")))
        )
        top_id = getattr(top_field, "id", None)
        if top_id is None:
            # If field id is missing, revert to idling all drones
            for c in components:
                environment.assign_group(c, "idle")
            return

        top_group = f"protecting {top_id}"

        # Field center for distance calculations
        left = getattr(top_field, "left", 0)
        right = getattr(top_field, "right", 0)
        top = getattr(top_field, "top", 0)
        bottom = getattr(top_field, "bottom", 0)
        top_center = ((left + right) / 2.0, (top + bottom) / 2.0)

        # Current drones protecting the top field
        current_top = [c for c in components if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) == top_id]

        required = int(getattr(top_field, "drones_for_full_protection", 0) or 0)

        final_group = {}

        # If already fully protected, assign current protectors to top_group and idle others
        if len(current_top) >= required:
            for c in components:
                if c in current_top:
                    final_group[c] = top_group
                else:
                    final_group[c] = "idle"
        else:
            # Preserve current protectors
            for c in current_top:
                final_group[c] = top_group

            need = max(0, required - len(current_top))

            # Build candidate pool: all drones not already protecting the top field
            candidates = [c for c in components if c not in current_top]

            # Sort candidates by distance to the top field center
            cand_with_dist = []
            for c in candidates:
                loc = getattr(c, "location", None)
                if loc is None or not hasattr(loc, "x") or not hasattr(loc, "y"):
                    dist = float("inf")
                else:
                    dist = math.hypot(loc.x - top_center[0], loc.y - top_center[1])
                cand_with_dist.append((dist, c))
            cand_with_dist.sort(key=lambda t: t[0])

            allocated = 0
            for dist, c in cand_with_dist:
                if allocated >= need:
                    break
                final_group[c] = top_group
                allocated += 1

            # Any remaining drones not allocated to the top field become idle
            for dist, c in cand_with_dist[allocated:]:
                final_group[c] = "idle"

        # Apply assignments
        # Ensure every drone is assigned to a known group
        for c in components:
            environment.assign_group(c, final_group.get(c, "idle"))
