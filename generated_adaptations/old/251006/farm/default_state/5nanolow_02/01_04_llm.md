import abc

# The base class is expected to be importable from the given path
from generated_adaptations.base_classes.farm import FarmAdaptation as _BaseFarmAdaptation

class SmartFarmAdaptation(_BaseFarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        """
        Assign drones into groups:
        - "idle": idle drones
        - "protecting {field_id}": drones protecting a specific field
        Strategy:
        - Identify the field with the highest threat_level > 0.
        - If that field is not fully protected, assign as many drones as needed to reach full protection.
        - Do not move drones that are already protecting this field.
        - All other drones should be assigned to 'idle' (single assignment per drone).
        """
        # Gather fields
        fields = list(environment.fields)

        # Current protection counts per field (drones currently protecting that field)
        current_protecting = {field.id: 0 for field in fields}
        for drone in components:
            if drone.state == "protecting" and drone.target_id is not None:
                tid = drone.target_id
                current_protecting[tid] = current_protecting.get(tid, 0) + 1

        # Identify best field to protect: highest threat_level > 0
        best_field = None
        best_threat = -1.0
        for field in fields:
            if field.threat_level > 0 and field.threat_level > best_threat:
                best_threat = field.threat_level
                best_field = field

        # If no field needs attention, assign everyone to idle and return
        if best_field is None:
            for drone in components:
                environment.assign_group(drone, "idle")
            return

        # How many drones are needed to fully protect this field?
        needed_total = getattr(best_field, "drones_for_full_protection", 0)
        current = current_protecting.get(best_field.id, 0)
        need = max(0, needed_total - current)

        # If already fully protected, ensure current drones stay protecting best_field
        if need == 0:
            # No reassignments needed; but ensure at least keep those currently protecting as is.
            # Drones not protecting best_field will be idle.
            for drone in components:
                if drone.state == "protecting" and drone.target_id == best_field.id:
                    environment.assign_group(drone, f"protecting {best_field.id}")
                else:
                    environment.assign_group(drone, "idle")
            return

        # Otherwise, we need to assign exactly 'need' additional drones to protect best_field.
        assigned_to_protect = 0

        for drone in components:
            # If this drone is already protecting the best field, keep it there (no re-assignment)
            if drone.state == "protecting" and drone.target_id == best_field.id:
                environment.assign_group(drone, f"protecting {best_field.id}")
                continue

            if assigned_to_protect < need:
                # Assign this drone to protect the best field
                environment.assign_group(drone, f"protecting {best_field.id}")
                assigned_to_protect += 1
            else:
                # Remaining drones go idle
                environment.assign_group(drone, "idle")

        # Note: This ensures each drone is assigned exactly once in this call.
