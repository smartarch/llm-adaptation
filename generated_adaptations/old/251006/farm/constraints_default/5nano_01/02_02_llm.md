import math
from generated_adaptations.base_classes.farm import FarmAdaptation

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        # Identify fields with positive threat
        fields_with_threat = [f for f in environment.fields if getattr(f, "threat_level", 0) > 0]

        # If no threats, idle all drones
        if not fields_with_threat:
            for d in components:
                environment.assign_group(d, "idle")
            return

        # Sort threat fields by threat level (descending)
        threat_fields = sorted(fields_with_threat, key=lambda f: getattr(f, "threat_level", 0), reverse=True)
        top_field = threat_fields[0]
        top_group = f"protecting {top_field.id}"

        # If the top field group is not valid, idle everything as a fallback
        if top_group not in group_ids:
            for d in components:
                environment.assign_group(d, "idle")
            return

        # Center of the top field
        cx = (top_field.left + top_field.right) / 2.0
        cy = (top_field.top + top_field.bottom) / 2.0

        # Drones required for full protection
        required_top = getattr(top_field, "drones_for_full_protection", 0)

        # Build final assignment mapping (drone -> group)
        final_group_for = {}

        # 1) Drones already protecting the top field stay in its group
        currently_top = [d for d in components if d.state == "protecting" and d.target_id == top_field.id]
        for d in currently_top:
            final_group_for[d] = top_group

        # 2) If more drones are needed for top field, assign closest available drones
        if required_top > len(currently_top):
            need = required_top - len(currently_top)
            candidates = []
            for d in components:
                if d in currently_top:
                    continue
                # Distance to top field center
                dist = float("inf")
                if getattr(d, "location", None) is not None:
                    dist = math.hypot(d.location.x - cx, d.location.y - cy)
                candidates.append((dist, d))
            candidates.sort(key=lambda t: t[0])
            for i in range(min(need, len(candidates))):
                drone = candidates[i][1]
                final_group_for[drone] = top_group

        # 3) Consider other threat fields in order, allocate if possible
        available = [d for d in components if d not in final_group_for]

        for fld in threat_fields[1:]:
            group = f"protecting {fld.id}"
            if group not in group_ids:
                continue
            currently = [d for d in final_group_for if final_group_for[d] == group]
            current_count = len(currently)
            needed_full = getattr(fld, "drones_for_full_protection", 0) - current_count
            if needed_full > 0 and available:
                fx = (fld.left + fld.right) / 2.0
                fy = (fld.top + fld.bottom) / 2.0
                cand = []
                for d in available:
                    dist = float("inf")
                    if getattr(d, "location", None) is not None:
                        dist = math.hypot(d.location.x - fx, d.location.y - fy)
                    cand.append((dist, d))
                cand.sort(key=lambda t: t[0])
                take = min(needed_full, len(cand))
                for i in range(take):
                    drone = cand[i][1]
                    final_group_for[drone] = group
                # remove taken
                taken = set([cand[i][1] for i in range(take)])
                available = [d for d in available if d not in taken]

        # 4) If there are still available drones, assign at least 1 to remaining threatened fields
        for fld in threat_fields[1:]:
            group = f"protecting {fld.id}"
            if group not in group_ids:
                continue
            if not available:
                break
            # If already have at least 1 drone for this field, skip
            currently = [d for d in final_group_for if final_group_for[d] == group]
            if len(currently) >= 1:
                continue
            fx = (fld.left + fld.right) / 2.0
            fy = (fld.top + fld.bottom) / 2.0
            dist_list = [
                (math.hypot(d.location.x - fx, d.location.y - fy) if getattr(d, "location", None) else float("inf"), d)
                for d in available
            ]
            dist_list.sort(key=lambda t: t[0])
            drone = dist_list[0][1]
            final_group_for[drone] = group
            available.remove(drone)

        # 5) Remaining drones idle
        for d in available:
            final_group_for[d] = "idle"

        # Ensure every drone has a final group
        for d in components:
            if d not in final_group_for:
                final_group_for[d] = "idle"

        # Apply a single assignment per drone
        for d in components:
            environment.assign_group(d, final_group_for[d])
