# Final refined adaptation strategy:
# - Select the most threatened field with a deterministic tie-breaker:
#   if multiple fields share the same maximum threat, pick the one with the largest numeric suffix
#   in the field id (e.g., Field_4 over Field_1).
# - Keep drones already protecting a field in their current protection group.
# - Allocate remaining drones to fully protect the top field first, then to the next most-threatened fields.
# - Ensure every drone is assigned exactly once.

from typing import List

from generated_adaptations.base_classes.farm import FarmAdaptation  # type: ignore

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def _field_sort_key(self, f):
        # primary: threat level (desc), secondary: numeric suffix of id (desc)
        threat = float(getattr(f, "threat_level", 0.0))
        fid = getattr(f, "id", "")
        # extract trailing number if present
        import re
        m = re.search(r'(\d+)$', str(fid))
        num = int(m.group(1)) if m else 0
        return (threat, num)

    def assign_drones(self, components: List, environment, group_ids: List[str], step: int):
        # Collect threatened fields and sort with tie-breaker
        fields_with_threat = [f for f in environment.fields if getattr(f, "threat_level", 0.0) > 0.0]
        fields_with_threat.sort(key=self._field_sort_key, reverse=True)

        if not fields_with_threat:
            for c in components:
                environment.assign_group(c, "idle")
            return

        # Map field_id -> current protectors count (only from drones currently protecting)
        current_protecting = {f.id: 0 for f in fields_with_threat}
        for c in components:
            if getattr(c, "state", None) == "protecting":
                fid = getattr(c, "target_id", None)
                if fid in current_protecting:
                    current_protecting[fid] += 1

        # Compute needs per field (considering only threatened fields)
        needs = {}
        for f in fields_with_threat:
            fid = f.id
            required = 0
            try:
                required = int(getattr(f, "drones_for_full_protection", 0))
            except Exception:
                required = 0
            needs[fid] = max(0, required - current_protecting.get(fid, 0))

        # Build assignment plan: ensure exactly one assignment per drone
        desired_groups = [None] * len(components)

        # Step A: keep drones already protecting in their current group
        for i, c in enumerate(components):
            if getattr(c, "state", None) == "protecting":
                fid = getattr(c, "target_id", None)
                if fid is not None:
                    desired_groups[i] = f"protecting {fid}"
                else:
                    desired_groups[i] = "idle"

        # Step B: allocate unassigned drones to top fields in threat order
        unassigned_indices = [i for i, g in enumerate(desired_groups) if g is None]

        for f in fields_with_threat:
            fid = f.id
            if needs.get(fid, 0) <= 0:
                continue
            while needs[fid] > 0 and unassigned_indices:
                idx = unassigned_indices.pop(0)
                desired_groups[idx] = f"protecting {fid}"
                needs[fid] -= 1

        # Step C: remaining drones become idle
        for idx in unassigned_indices:
            desired_groups[idx] = "idle"

        # Apply assignments
        for c, grp in zip(components, desired_groups):
            environment.assign_group(c, grp)
