import abc

# The base class is expected to be importable from the given path
from generated_adaptations.base_classes.farm import FarmAdaptation as _BaseFarmAdaptation

class SmartFarmAdaptation(_BaseFarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        """
        Robust single-pass assignment:
        - Protect the field with the highest threat_level (>0) by allocating drones until its
          protection is full (drones_for_full_protection). Preserve drones already protecting it.
        - All other drones are assigned to idle in this decision.
        - Ensure each drone is assigned exactly once in this call.
        """
        # Gather fields
        fields = list(environment.fields)

        # Current protection per field (count drones currently protecting that field)
        current_protecting = {field.id: 0 for field in fields}
        for d in components:
            if d.state == "protecting" and d.target_id is not None:
                current_protecting[d.target_id] = current_protecting.get(d.target_id, 0) + 1

        # Find the best field to defend: highest threat_level > 0
        best_field = None
        best_threat = -1.0
        for f in fields:
            if f.threat_level > 0 and f.threat_level > best_threat:
                best_threat = f.threat_level
                best_field = f

        # If no field needs attention, idle all
        if best_field is None:
            for d in components:
                environment.assign_group(d, "idle")
            return

        # How many drones are needed to reach full protection for best_field?
        needed_total = getattr(best_field, "drones_for_full_protection", 0)
        current = current_protecting.get(best_field.id, 0)
        need = max(0, needed_total - current)

        # Build final assignment for each drone (exactly one group per drone)
        final_group_map = {}

        if need == 0:
            # Field already fully protected: keep current protecting drones, idle others
            for d in components:
                if d.state == "protecting" and d.target_id == best_field.id:
                    final_group_map[d] = f"protecting {best_field.id}"
                else:
                    final_group_map[d] = "idle"
        else:
            # First, keep drones already protecting this field
            for d in components:
                if d.state == "protecting" and d.target_id == best_field.id:
                    final_group_map[d] = f"protecting {best_field.id}"
            # Then assign additional drones up to 'need'
            remaining_needed = need
            for d in components:
                if remaining_needed <= 0:
                    break
                if d in final_group_map:
                    continue
                final_group_map[d] = f"protecting {best_field.id}"
                remaining_needed -= 1
            # Any drones not yet assigned go idle
            for d in components:
                if d not in final_group_map:
                    final_group_map[d] = "idle"

        # Now apply the assignments (exactly one call per drone)
        for d in components:
            environment.assign_group(d, final_group_map[d])
