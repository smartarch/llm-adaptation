"""
Strategy reasoning (embedded as comments for traceability):
- The goal is to protect fields from birds by assigning drones to field-specific protection groups.
- Always fully protect the field with the highest threat level first, using the closest available drones.
- If that field already has enough drones protecting it, keep those drones in place. Then use the remaining drones to
  protect other fields (in descending order of threat) as much as possible, until drones run out.
- For each field with threat_level > 0, we have a group named "protecting {field.id}".
- All drones must end up assigned to exactly one group: either "idle" or one of the "protecting {field.id}" groups.
- We respect current protection by keeping drones that are already protecting a field (when that field is in the sequence)
  unless we run out of drones to fulfill higher-priority protections.
- The distance heuristic (to a field center) is used to pick the closest drones to fill a field's protection need.
"""

import math
from generated_adaptations.base_classes import farm as base_farm

class SmartFarmAdaptation(base_farm.FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        # Gather fields with positive threat level
        fields_with_threat = [f for f in environment.fields if getattr(f, "threat_level", 0) > 0]

        # If no field needs protection, idle all drones
        if not fields_with_threat:
            for c in components:
                environment.assign_group(c, "idle")
            return

        # Sort fields by threat level (highest first)
        fields_desc = sorted(fields_with_threat, key=lambda fld: getattr(fld, "threat_level", 0), reverse=True)

        # Helper: compute center of a field
        def field_center(field):
            left = getattr(field, "left", 0)
            right = getattr(field, "right", 0)
            top = getattr(field, "top", 0)
            bottom = getattr(field, "bottom", 0)
            return ((left + right) / 2.0, (top + bottom) / 2.0)

        # Build a quick mapping of current protectors per field
        current_by_field = {}
        for f in fields_desc:
            fid = getattr(f, "id", None)
            if fid is None:
                continue
            current_by_field[fid] = [c for c in components if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) == fid]

        # Prepare final assignments per drone (one group per drone)
        assigned = set()
        final_group = {}

        # Step 1: Top field - ensure it's fully protected using closest drones
        top_field = fields_desc[0]
        top_id = getattr(top_field, "id", None)
        top_group = f"protecting {top_id}"
        top_center = field_center(top_field)

        # Keep current protectors for the top field in place first
        for d in current_by_field.get(top_id, []):
            final_group[d] = top_group
            assigned.add(d)

        # Compute how many more drones are needed for full protection
        required_top = int(getattr(top_field, "drones_for_full_protection", 0) or 0)
        need_top = max(0, required_top - len(current_by_field.get(top_id, [])))
        if need_top > 0:
            # Build candidate drones not yet assigned
            candidates = [c for c in components if c not in assigned]
            # Distances to the top field center
            cand_with_dist = []
            for c in candidates:
                loc = getattr(c, "location", None)
                if loc is None or not hasattr(loc, "x") or not hasattr(loc, "y"):
                    dist = float("inf")
                else:
                    dist = math.hypot(loc.x - top_center[0], loc.y - top_center[1])
                cand_with_dist.append((dist, c))
            cand_with_dist.sort(key=lambda t: t[0])

            allocated = 0
            for dist, c in cand_with_dist:
                if allocated >= need_top:
                    break
                final_group[c] = top_group
                assigned.add(c)
                allocated += 1

        # Step 2: Other fields (in threat-descending order)
        for field in fields_desc[1:]:
            fid = getattr(field, "id", None)
            if fid is None:
                continue
            group = f"protecting {fid}"
            center = field_center(field)

            # Keep current protectors for this field in place
            current_list = current_by_field.get(fid, [])
            for d in current_list:
                if d not in assigned:
                    final_group[d] = group
                    assigned.add(d)

            required = int(getattr(field, "drones_for_full_protection", 0) or 0)
            need = max(0, required - len(current_list))
            if need > 0:
                # Build candidate drones not yet assigned
                candidates = [c for c in components if c not in assigned]
                cand_with_dist = []
                for c in candidates:
                    loc = getattr(c, "location", None)
                    if loc is None or not hasattr(loc, "x") or not hasattr(loc, "y"):
                        dist = float("inf")
                    else:
                        dist = math.hypot(loc.x - center[0], loc.y - center[1])
                    cand_with_dist.append((dist, c))
                cand_with_dist.sort(key=lambda t: t[0])

                allocated = 0
                for dist, c in cand_with_dist:
                    if allocated >= need:
                        break
                    final_group[c] = group
                    assigned.add(c)
                    allocated += 1

        # Step 3: Any drones not assigned yet become idle
        for c in components:
            if c not in assigned:
                final_group[c] = "idle"

        # Apply all assignments (one per drone)
        for c in components:
            environment.assign_group(c, final_group.get(c, "idle"))
