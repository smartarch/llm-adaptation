Reasoning and updated adaptation strategy:
- The previous approach protected only the single highest-threat field, which left the other fields vulnerable and yielded low overall protection coverage and higher damage.
- Improved strategy: distribute drones to protect multiple fields in a prioritized way, while still prioritizing the most dangerous areas. Specifically:
  - Consider all fields with threat_level > 0 and sort them by threat level (descending).
  - For each field in that order, compute how many drones are needed to reach full protection: need = drones_for_full_protection - current_protecting_drones. If need > 0, allocate up to that many drones from the remaining pool.
  - To minimize travel time and avoid unnecessary moves, allocate drones by choosing the closest available drones to the field center first.
  - If we have more drones than total needs, remaining drones stay idle (or can be kept in their current protective state if they already protect a field with high threat, but we explicitly reassign to idle to be explicit per requirements).
  - If no field has threat (>0), assign all drones to idle.
- This approach increases protection coverage across high-threat fields (not just the top one) and reduces the likelihood of large damage swings when threats shift between fields.

from typing import List
import math

# Import the base class from the expected location
from generated_adaptations.base_classes.farm import FarmAdaptation

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
    
    def _field_center(self, field) -> tuple:
        cx = (field.left + field.right) / 2.0
        cy = (field.top + field.bottom) / 2.0
        return (cx, cy)
    
    def _distance(self, p, q) -> float:
        dx = p.x - q[0]
        dy = p.y - q[1]
        return math.hypot(dx, dy)
    
    def assign_drones(self, components: List, environment, group_ids: List[str], step: int):
        # Gather all fields with threat > 0 and sort by threat level descending
        fields_with_threat = [f for f in environment.fields if getattr(f, "threat_level", 0) > 0]
        if not fields_with_threat:
            for comp in components:
                environment.assign_group(comp, "idle")
            return
        
        fields_with_threat.sort(key=lambda f: f.threat_level, reverse=True)
        
        # Prepare a mapping for quick access (optional)
        # We will allocate drones progressively to each field in order
        field_needs = {}
        for field in fields_with_threat:
            needed_total = getattr(field, "drones_for_full_protection", 0)
            current_protecting = getattr(field, "protecting_drones", 0)
            need = int(needed_total - current_protecting)
            if need > 0:
                field_needs[field] = need
        
        # If nothing needs protection, keep existing protections explicit or idle
        if not field_needs:
            for comp in components:
                # If a drone is already protecting some field with high threat, keep it; otherwise idle
                kept = False
                for field in fields_with_threat:
                    if getattr(comp, "state", None) == "protecting" and getattr(comp, "target_id", None) == field.id:
                        environment.assign_group(comp, f"protecting {field.id}")
                        kept = True
                        break
                if not kept:
                    environment.assign_group(comp, "idle")
            return
        
        # Sort candidate drones by distance to the field centers as we allocate across fields
        # We'll build a pool of available drones
        assigned_to_field = {}
        unassigned_drones = list(components)
        # For deterministic behavior, process fields in threat order
        for field, need in field_needs.items():
            if not unassigned_drones:
                break
            center = self._field_center(field)
            # Compute distances for remaining drones
            cand = []
            for comp in unassigned_drones:
                pos = getattr(comp, "location", None)
                dist = float('inf')
                if pos is not None:
                    dist = self._distance(pos, center)
                cand.append((dist, comp))
            cand.sort(key=lambda x: x[0])
            # Take the closest 'need' drones
            take = min(need, len(cand))
            for i in range(take):
                dist, comp = cand[i]
                environment.assign_group(comp, f"protecting {field.id}")
                assigned_to_field.setdefault(field, []).append(comp)
            # Remove assigned drones from pool
            assigned_set = set(assigned_to_field.get(field, []))
            unassigned_drones = [c for c in unassigned_drones if c not in assigned_set]
        
        # Remaining drones (not assigned to any field for protection) go idle
        for comp in unassigned_drones:
            environment.assign_group(comp, "idle")
