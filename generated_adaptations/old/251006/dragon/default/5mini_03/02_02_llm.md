"""
SmartAdaptation (revised)

Reasoning and analysis:
- Problem observed: previous strategy never spawned new villagers in practice because it
  kept a minimum farmer farming (1) even when only 2 farmers existed; that prevented forming
  a spawn pair and thus blocked all future spawns. With no warriors spawned, no sustained
  attacking happened and the Dragon survived.
- Goal: kill the Dragon fast. Warriors are the main damage dealers; farmers produce wheat
  needed to spawn more villagers. We must balance immediate warrior production with sustaining
  wheat income.
- Constraints from the prompt:
  - All Warriors should go to the Cave and attack the Dragon.
  - All Farmers should stay in the Village and either farm or be assigned to spawn groups.
  - Each component must be assigned to exactly one group each step.

Adaptation strategy (improvements):
1) assign_in_cave:
   - Always send Warriors to "attack".
   - Send Farmers back to "village" so they continue producing wheat / participating in spawns.

2) assign_in_village (improved spawn logic):
   - Send any Warriors in Village to the Cave.
   - For Farmers, form spawn pairs when wheat permits. Unlike the previous version, allow
     spawning even if it leaves zero farmers farming when the total number of farmers is small:
     - If there are 2 or fewer farmers, we allow using both to form a spawn pair (min_farmers_farming = 0).
     - If there are more than 2 farmers, keep at least 1 farming to maintain income (min_farmers_farming = 1).
   - Spawn priority:
     - If wheat >= 12 and (Dragon HP is still high or we have surplus wheat), prefer spawning Warriors.
     - Otherwise, if wheat >= 10, spawn Farmers to grow long-term wheat income.
   - Iteratively assign farmers in pairs to spawn groups until we can't form more pairs or run out of wheat,
     then assign any remaining farmers to "farm".
   - This change allows early spawns (critical when starting with few villagers) and grows the economy
     early so subsequent warrior spawns become possible.

Implementation notes:
- All reasoning and description included in this docstring.
- The class implements the required interface and uses environment.assign_group(component, group_id)
  with exact group names requested.

"""

from generated_adaptations.base_classes.dragon import DragonHuntAdaptation


class SmartAdaptation(DragonHuntAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_in_village(self, components, environment, group_ids, step: int):
        # Exact group names required by the environment
        GROUP_FARM = "farm"
        GROUP_CAVE = "cave"
        GROUP_SPAWN_FARMER = "spawn farmer"
        GROUP_SPAWN_WARRIOR = "spawn warrior"

        # Partition villagers by role
        farmers = [c for c in components if getattr(c, "role", "") == "Farmer"]
        warriors = [c for c in components if getattr(c, "role", "") == "Warrior"]

        # Send all village warriors to the cave immediately
        for w in warriors:
            environment.assign_group(w, GROUP_CAVE)

        # Read wheat and dragon hp safely
        try:
            wheat = int(environment.farm.wheat)
        except Exception:
            wheat = 0
        try:
            dragon_hp = int(environment.dragon.hp)
        except Exception:
            dragon_hp = 0

        # Decide minimum farmers to leave farming:
        # - If we have only 2 or fewer farmers, allow using both for an immediate spawn (min_farmers_farming = 0).
        # - If we have more than 2 farmers, keep at least 1 farming to maintain income.
        total_farmers = len(farmers)
        if total_farmers <= 2:
            min_farmers_farming = 0
        else:
            min_farmers_farming = 1

        # Prepare mutable list of farmers to allocate
        remaining = list(farmers)

        # Helper to pop two farmers for spawning (returns list of two or None)
        def pop_two():
            if len(remaining) >= 2:
                return [remaining.pop(0), remaining.pop(0)]
            return None

        assigned_spawn = []  # tuples (component, group)

        # While we can form pairs beyond the reserved farmers, attempt to spawn
        # Allow forming pairs as long as after removing pairs we still have >= min_farmers_farming
        # or if min_farmers_farming == 0 we can use all farmers.
        while True:
            # How many farmers would remain if we took a pair now?
            if len(remaining) < 2:
                break
            # If taking a pair would violate the min_farmers_farming constraint, stop
            if len(remaining) - 2 < min_farmers_farming:
                # Special-case: if min_farmers_farming == 0, we can still take the pair
                if min_farmers_farming > 0:
                    break

            # Decide what to spawn for this pair
            # Prefer spawning a warrior if we have enough wheat AND:
            # - dragon hp is relatively high (we need damage), OR
            # - we have surplus wheat (>=24) to allow multiple warriors.
            spawn_warrior_preferred = False
            if wheat >= 12:
                if dragon_hp > 20 or wheat >= 24:
                    spawn_warrior_preferred = True

            if spawn_warrior_preferred and wheat >= 12:
                pair = pop_two()
                if pair is None:
                    break
                for p in pair:
                    assigned_spawn.append((p, GROUP_SPAWN_WARRIOR))
                wheat -= 12
                continue

            # Otherwise, if we can spawn a farmer, do so to boost wheat income
            if wheat >= 10:
                pair = pop_two()
                if pair is None:
                    break
                for p in pair:
                    assigned_spawn.append((p, GROUP_SPAWN_FARMER))
                wheat -= 10
                continue

            # Can't afford any spawn or prefer not to; stop spawning loop
            break

        # Assign spawn participants to their groups
        for comp, grp in assigned_spawn:
            environment.assign_group(comp, grp)

        # Assign any remaining farmers to farm
        for f in remaining:
            environment.assign_group(f, GROUP_FARM)

    def assign_in_cave(self, components, environment, group_ids, step: int):
        # Exact group names required by the environment
        GROUP_ATTACK = "attack"
        GROUP_CAVE = "cave"
        GROUP_VILLAGE = "village"

        # Warriors attack; farmers return to village to farm or spawn
        for c in components:
            if getattr(c, "role", "") == "Warrior":
                environment.assign_group(c, GROUP_ATTACK)
            else:
                environment.assign_group(c, GROUP_VILLAGE)
