# SmartFarmAdaptation
# 
# Reasoning and strategy (described here as comments per instructions):
# 
# Goal:
#   Assign drones to groups so the field with the highest threat_level is fully protected
#   using the closest drones. If the field is already fully protected, keep the drones
#   assigned there. All other drones should be idle.
#
# Key points:
# - We must choose the single field with the maximum threat_level among fields with
#   threat_level > 0. Ties are broken deterministically by field.id (lexicographic).
# - Full protection requires field.drones_for_full_protection drones. If the number of
#   currently protecting drones (state == "protecting" and target_id == field.id)
#   meets or exceeds this requirement, we leave those drones in place (assign them to
#   the protecting group). Otherwise, we choose the closest other drones (by Euclidean
#   distance to the field center) to reach the required count.
# - All drones not assigned to the chosen field's protecting group will be assigned to
#   the "idle" group.
# - Every component must be explicitly assigned to a group via environment.assign_group.
#
# Implementation notes:
# - We treat drones currently in state "protecting" and targeting the chosen field as
#   already committed and keep them there.
# - For required additional drones, we consider all other drones (idle, moving_to_field,
#   protecting other fields) and pick the nearest ones to the chosen field center.
# - Use math.ceil when interpreting drones_for_full_protection to be safe in case it's
#   not integral.
# - We validate group names against the provided group_ids list; if a desired group is
#   missing, we fall back to "idle" where appropriate.
#
# The class below implements the strategy and uses environment.assign_group(component, group_id)
# to perform the assignments as required by the adaptation API.

import math
from generated_adaptations.base_classes.farm import FarmAdaptation

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        # Helper to compute squared distance between drone and field center
        def dist2(comp, center):
            dx = comp.location.x - center[0]
            dy = comp.location.y - center[1]
            return dx * dx + dy * dy

        # Find fields with positive threat level
        candidate_fields = [f for f in environment.fields if getattr(f, "threat_level", 0) > 0]

        # Default: assign all to idle if no field requires protection
        idle_group = "idle"
        if idle_group not in group_ids:
            # If "idle" somehow missing, attempt to use any provided group or do nothing
            # but per contract "idle" should be present. We'll handle gracefully.
            fallback = group_ids[0] if group_ids else None
            for comp in components:
                if fallback is not None:
                    environment.assign_group(comp, fallback)
            return

        if not candidate_fields:
            # No threats: assign all drones to idle
            for comp in components:
                environment.assign_group(comp, idle_group)
            return

        # Choose the field with highest threat_level; tie-break by field.id
        candidate_fields.sort(key=lambda f: (-f.threat_level, str(f.id)))
        field = candidate_fields[0]
        protect_group = f"protecting {field.id}"
        if protect_group not in group_ids:
            # If the expected protecting group is not present, assign everyone idle
            for comp in components:
                environment.assign_group(comp, idle_group)
            return

        # Compute field center
        center_x = (field.left + field.right) / 2.0
        center_y = (field.top + field.bottom) / 2.0
        center = (center_x, center_y)

        # Determine how many drones needed for full protection
        try:
            required = int(math.ceil(field.drones_for_full_protection))
        except Exception:
            # Fallback: if attribute missing or invalid, assume 1
            required = 1

        # Identify drones already protecting this field
        protecting_now = []
        others = []
        for comp in components:
            if comp.state == "protecting" and comp.target_id == field.id:
                protecting_now.append(comp)
            else:
                others.append(comp)

        # Number still needed
        need = max(0, required - len(protecting_now))

        # Rank other drones by distance to the field center
        others_sorted = sorted(others, key=lambda c: dist2(c, center))

        # Select the closest 'need' drones to join protection
        selected_for_protection = set(others_sorted[:need])

        # Assign groups:
        # - Drones in protecting_now -> protecting group
        # - Drones in selected_for_protection -> protecting group
        # - All others -> idle
        for comp in components:
            if comp in protecting_now or comp in selected_for_protection:
                environment.assign_group(comp, protect_group)
            else:
                environment.assign_group(comp, idle_group)
