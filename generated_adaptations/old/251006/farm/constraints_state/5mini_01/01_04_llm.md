Reasoning and adaptation strategy

- Problem found from tests: leaving too many drones idle when they could be used to fully protect additional fields. The tests expect the controller to not only fully protect the top-threat field, but also try to fully protect other threatened fields using remaining drones.
- Strategy:
  1. Sort fields with threat_level > 0 by descending threat_level.
  2. Process fields in that order. For each field:
     - Count drones already committed to it (state "protecting" or "moving_to_field" with target_id == field.id). Those committed drones count toward the field's required drones_for_full_protection and are kept reserved.
     - If committed >= required, keep those drones assigned to that field.
     - If committed < required, pick the closest drones from the pool of currently unreserved drones (this pool may include drones protecting lower-priority fields unless those lower-priority fields were already fully reserved) until the required number is reached (or no drones remain).
     - Reserve these selected drones for that field.
  3. After attempting to fully protect fields in priority order, assign any remaining (unreserved) drones to "idle".
  4. Always explicitly assign every drone to a group and only use group ids from group_ids (fall back defensively if needed).

This preserves the rule to always ensure the highest threat field is fully protected using the closest drones, keeps fully protected fields' drones in place, and uses leftover drones to protect additional fields where possible to reduce idle drones.

from math import hypot
from generated_adaptations.base_classes.farm import FarmAdaptation

class SmartFarmAdaptation(FarmAdaptation):
    def assign_drones(self, components, environment, group_ids, step: int):
        """
        Greedy multi-field protection strategy:
        - Sort threatened fields by descending threat_level.
        - For each field, keep its committed drones (protecting + moving_to_field to that field).
        - If more drones are needed, pick closest available drones to fully protect it.
        - After processing fields, assign all remaining drones to "idle".
        - Always explicitly assign every drone to a valid group.
        """
        # Helper for protecting group name
        def protecting_group_name(field_id):
            return f"protecting {field_id}"

        # Build list of threatened fields (threat_level > 0)
        threatened = [f for f in environment.fields if getattr(f, "threat_level", 0) > 0]
        if not threatened:
            # No threats: assign everyone to idle if possible
            for comp in components:
                if "idle" in group_ids:
                    environment.assign_group(comp, "idle")
                else:
                    # Defensive fallback
                    if group_ids:
                        environment.assign_group(comp, group_ids[0])
            return

        # Sort fields by descending threat_level
        threatened.sort(key=lambda f: getattr(f, "threat_level", 0), reverse=True)

        # Prepare selection tracking
        reserved_set = set()  # store id(component) of drones reserved for protection
        assignment_map = {}   # comp -> group_name to assign at the end

        # Precompute drone list for convenience
        comps = list(components)

        # For distance computations to field center
        def field_center(field):
            fx = (getattr(field, "left", 0) + getattr(field, "right", 0)) / 2.0
            fy = (getattr(field, "top", 0) + getattr(field, "bottom", 0)) / 2.0
            return fx, fy

        # For each threatened field in priority order, reserve drones to fully protect it
        for field in threatened:
            required = int(getattr(field, "drones_for_full_protection", 0))
            if required <= 0:
                continue

            protect_name = protecting_group_name(field.id)
            # Only attempt to protect if protecting group exists; otherwise skip field.
            if protect_name not in group_ids:
                continue

            fx, fy = field_center(field)

            # Find drones currently committed to this field (protecting or moving_to_field with matching target_id)
            committed = []
            for c in comps:
                if id(c) in reserved_set:
                    # already reserved for a higher-priority field
                    continue
                state = getattr(c, "state", None)
                target = getattr(c, "target_id", None)
                if (state == "protecting" or state == "moving_to_field") and target == field.id:
                    committed.append(c)

            # Reserve committed drones first
            for c in committed:
                if id(c) not in reserved_set:
                    reserved_set.add(id(c))
                    assignment_map[c] = protect_name

            committed_count = len(committed)
            need = max(0, required - committed_count)
            if need <= 0:
                # already fully protected
                continue

            # Build list of available drones (not yet reserved)
            available = [c for c in comps if id(c) not in reserved_set]
            # Sort available drones by distance to field center (closest first)
            available.sort(key=lambda c: hypot(getattr(c.location, "x", 0) - fx, getattr(c.location, "y", 0) - fy))

            # Select as many as needed (or as many as available)
            for c in available[:need]:
                reserved_set.add(id(c))
                assignment_map[c] = protect_name

        # After attempting to protect fields, assign unreserved drones to idle (if available)
        idle_valid = "idle" in group_ids
        fallback_group = group_ids[0] if group_ids else None

        for c in comps:
            if c in assignment_map:
                # already assigned to a protecting group
                continue
            # assign to idle if possible, else fallback to first group
            if idle_valid:
                assignment_map[c] = "idle"
            elif fallback_group is not None:
                assignment_map[c] = fallback_group
            else:
                # No group available (shouldn't happen), skip
                assignment_map[c] = None

        # Now perform assignments explicitly (only to valid groups)
        for c, g in assignment_map.items():
            if g is None:
                continue
            # Defensive: ensure group exists in group_ids, else fallback to idle or first
            if g not in group_ids:
                if idle_valid:
                    environment.assign_group(c, "idle")
                elif fallback_group is not None:
                    environment.assign_group(c, fallback_group)
                else:
                    # Nothing valid to assign; skip
                    continue
            else:
                environment.assign_group(c, g)
