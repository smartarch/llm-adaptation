# One-pass, deterministic strategy
# - Always identify the most threatened field (threat_level > 0, highest).
# - If that field needs drones to reach its full_protection, assign unallocated drones
#   to protect it, one drone per iteration.
# - Drones already protecting a field stay with that field.
# - If no field needs protection or there are no threats, all drones become idle.

from typing import List

from generated_adaptations.base_classes.farm import FarmAdaptation  # type: ignore

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components: List, environment, group_ids: List[str], step: int):
        # Step 1: identify the single most threatened field
        best_field = None
        best_threat = -1.0
        for f in environment.fields:
            try:
                t = float(getattr(f, "threat_level", 0.0))
            except Exception:
                t = 0.0
            if t > best_threat and t > 0.0:
                best_threat = t
                best_field = f

        if best_field is None:
            # No threats: idle all drones
            for c in components:
                environment.assign_group(c, "idle")
            return

        field_id = best_field.id
        # Step 2: count how many drones are currently protecting the best field
        current_protecting = 0
        for c in components:
            if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) == field_id:
                current_protecting += 1

        # Step 3: how many drones are required for full protection of the best field
        required = getattr(best_field, "drones_for_full_protection", 0)
        try:
            required = int(required)
        except Exception:
            required = 0
        needed = max(0, required - current_protecting)

        # Step 4: assign drones in a single pass
        for c in components:
            state = getattr(c, "state", None)
            if state == "protecting":
                t_id = getattr(c, "target_id", None)
                if t_id is not None:
                    environment.assign_group(c, f"protecting {t_id}")
                else:
                    environment.assign_group(c, "idle")
                continue

            if needed > 0:
                environment.assign_group(c, f"protecting {field_id}")
                needed -= 1
            else:
                environment.assign_group(c, "idle")
