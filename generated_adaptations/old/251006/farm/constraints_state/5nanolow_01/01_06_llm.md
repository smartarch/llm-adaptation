from __future__ import annotations
import math

from generated_adaptations.base_classes.farm import FarmAdaptation  # type: ignore


class SmartFarmAdaptation(FarmAdaptation):
    def assign_drones(self, components, environment, group_ids, step: int):
        # Step 1: Identify the field with the highest threat level (> 0)
        target_field = None
        max_threat = -1.0
        for field in environment.fields:
            thr = getattr(field, "threat_level", 0.0)
            if thr > 0 and thr > max_threat:
                max_threat = thr
                target_field = field

        # If no field is threatened, assign all drones to idle
        if target_field is None:
            for c in components:
                environment.assign_group(c, "idle")
            return

        target_group = f"protecting {target_field.id}"
        drones_for_full = getattr(target_field, "drones_for_full_protection", 0)

        # Step 2: Count currently protecting drones for the target field
        current_protecting = 0
        for c in components:
            if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) == target_field.id:
                current_protecting += 1

        # Step 3: Determine how many more drones are needed to reach full protection
        needed_for_target = max(0, drones_for_full - current_protecting)

        # Step 4: Prepare final assignments (one group per drone)
        final_group_for = {c: "idle" for c in components}

        # a) Ensure currently protecting drones stay in the target group
        for c in components:
            if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) == target_field.id:
                final_group_for[c] = target_group

        # b) If more drones are needed for the target, pick closest ones
        if needed_for_target > 0:
            center_x = (getattr(target_field, "left", 0) + getattr(target_field, "right", 0)) / 2.0
            center_y = (getattr(target_field, "top", 0) + getattr(target_field, "bottom", 0)) / 2.0

            def dist_to_field(drone):
                loc = getattr(drone, "location", None)
                if loc is None:
                    return float("inf")
                dx = getattr(loc, "x", 0.0) - center_x
                dy = getattr(loc, "y", 0.0) - center_y
                return math.hypot(dx, dy)

            candidates = [
                d for d in components
                if not (getattr(d, "state", None) == "protecting" and getattr(d, "target_id", None) == target_field.id)
            ]
            candidates.sort(key=dist_to_field)

            for i in range(min(needed_for_target, len(candidates))):
                final_group_for[candidates[i]] = target_group

        # Step 5: After filling the top field, try to assist other threatened fields with remaining drones
        # Build a list of other fields with threat > 0
        remaining_candidates = [d for d in components if final_group_for[d] == "idle"]
        # Sort fields by threat (descending) to prioritize higher threats
        other_fields = [
            f for f in environment.fields
            if getattr(f, "threat_level", 0) > 0 and f.id != target_field.id
        ]
        other_fields.sort(key=lambda f: getattr(f, "threat_level", 0), reverse=True)

        for field in other_fields:
            # How many drones are currently protecting this field?
            current = sum(
                1 for d in components
                if getattr(d, "state", None) == "protecting" and getattr(d, "target_id", None) == field.id
            )
            need = max(0, getattr(field, "drones_for_full_protection", 0) - current)
            if need <= 0:
                continue

            # Compute center for this field
            cx = (getattr(field, "left", 0) + getattr(field, "right", 0)) / 2.0
            cy = (getattr(field, "top", 0) + getattr(field, "bottom", 0)) / 2.0

            def dist_to_field_by_field(drone):
                loc = getattr(drone, "location", None)
                if loc is None:
                    return float("inf")
                dx = getattr(loc, "x", 0.0) - cx
                dy = getattr(loc, "y", 0.0) - cy
                return math.hypot(dx, dy)

            # Recompute list of available drones excluding those already assigned
            available = [d for d in remaining_candidates if final_group_for[d] == "idle"]
            available.sort(key=dist_to_field_by_field)

            for i in range(min(need, len(available))):
                final_group_for[available[i]] = f"protecting {field.id}"
                # Remove from further consideration
                remaining_candidates.remove(available[i])

        # Step 6: Apply the final single assignment for every drone
        for c, grp in final_group_for.items():
            environment.assign_group(c, grp)
