import abc
from generated_adaptations.base_classes.farm import FarmAdaptation

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        """
        Improved strategy (more conservative moves):
        - Always try to fully protect the most-threatened field (top_field) using its protect_group.
        - Do not move drones away from other fields unless there are idle drones available.
        - After topping off the top_field, allocate any remaining idle drones to the next most-threatened
          fields in order, but only using drones that are currently idle (not protecting another field).
        - Any drones that cannot be assigned to a protecting group stay idle.
        - This approach increases protection on the top field with minimal drone movement.
        """
        # Collect fields with threat
        fields_with_threat = [f for f in environment.fields if getattr(f, "threat_level", 0.0) > 0]
        if not fields_with_threat:
            for c in components:
                environment.assign_group(c, "idle")
            return

        # Sort by threat level (desc) for determinism
        fields_with_threat.sort(key=lambda f: getattr(f, "threat_level", 0.0), reverse=True)

        top_field = fields_with_threat[0]
        top_group_id = f"protecting {top_field.id}"
        if top_group_id not in group_ids:
            for c in components:
                environment.assign_group(c, "idle")
            return

        # Determine required drones for full protection
        required_top = getattr(top_field, "drones_for_full_protection", None)
        if required_top is None:
            required_top = len(components)
        try:
            required_top = int(required_top)
        except Exception:
            try:
                required_top = int(float(required_top))
            except Exception:
                required_top = len(components)

        # Step 1: Reuse drones already protecting the top field
        assigned = set()
        current_top = 0
        for c in components:
            if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) == top_field.id:
                environment.assign_group(c, top_group_id)
                assigned.add(c)
                current_top += 1

        # Step 2: If needed, move idle drones to top_field to reach full protection
        if current_top < required_top:
            needed = required_top - current_top
            for c in components:
                if c in assigned:
                    continue
                # Only move drones that are idle (not currently protecting any field)
                if getattr(c, "state", None) == "idle" or not any(
                    getattr(d, "state", None) == "protecting" for d in components
                ):
                    environment.assign_group(c, top_group_id)
                    assigned.add(c)
                    current_top += 1
                    needed -= 1
                    if needed <= 0:
                        break

        # Step 3: Allocate remaining drones to next-high-threat fields using only idle drones
        # Build a pool of truly idle drones (not protecting any field)
        pool = [c for c in components if all(
            getattr(d, "state", None) != "protecting" or getattr(d, "target_id", None) != getattr(top_field, "id", None)  # harmless check
            for d in [c]
        )]
        # Counter-check: ensure pool contains drones not currently protecting any field
        pool = [c for c in components if not any(getattr(d, "state", None) == "protecting" for d in [c])]

        # Consider next fields in threat order, but only use idle drones
        for field in fields_with_threat[1:]:
            g = f"protecting {field.id}"
            if g not in group_ids:
                continue
            req = getattr(field, "drones_for_full_protection", None)
            if req is None:
                continue
            try:
                req = int(req)
            except Exception:
                try:
                    req = int(float(req))
                except Exception:
                    continue

            # Current protecting for this field
            current = 0
            for c in components:
                if getattr(c, "state", None) == "protecting" and getattr(c, "target_id", None) == field.id:
                    current += 1

            to_add = max(0, req - current)
            while to_add > 0 and pool:
                c = pool.pop()
                environment.assign_group(c, g)
                assigned.add(c)
                to_add -= 1

        # Step 4: Any remaining drones in pool go idle
        for c in pool:
            environment.assign_group(c, "idle")
