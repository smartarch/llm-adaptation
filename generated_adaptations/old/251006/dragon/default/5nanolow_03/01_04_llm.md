import abc
from typing import List
from generated_adaptations.base_classes.dragon import DragonHuntAdaptation

class SmartAdaptation(DragonHuntAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_in_village(self, components, environment, group_ids, step: int):
        """
        Divide Villagers in the Village into:
        - "farm": stay in Village and farm
        - "cave": go to Cave (to be attacked later)
        - "spawn farmer": for every two villagers assigned here and 10 wheat, spawn a Farmer
        - "spawn warrior": for every two villagers assigned here and 12 wheat, spawn a Warrior
        """
        # Initialize by assuming all Farmers stay in farm
        farmers = [c for c in components if getattr(c, "role", None) == "Farmer"]
        warriors = [c for c in components if getattr(c, "role", None) == "Warrior"]

        # Step 1: assign all farmers to "farm" by default
        for c in farmers:
            environment.assign_group(c, "farm")

        # Step 2: Warriors should go to cave to attack
        for w in warriors:
            environment.assign_group(w, "cave")

        # Step 3: Spawning decisions based on available wheat and farmers not already allocated to spawns
        # Compute current wheat
        wheat = 0
        if hasattr(environment, "farm") and hasattr(environment.farm, "wheat"):
            wheat = environment.farm.wheat

        # Determine how many farmers we can allocate to spawn groups
        # We will take from the pool of farmers not already allocated to spawns (currently all farmers
        # are in farm). We'll select from the list in order.
        remaining_for_spawning = [c for c in farmers]  # all farmers are available
        pick_index = 0

        # Spawn farmers: need 2 villagers and 10 wheat per spawn
        max_possible_farm_spawns = wheat // 10
        s_f = min(len(remaining_for_spawning) // 2, max_possible_farm_spawns)

        if s_f > 0:
            pick_count = 2 * s_f
            for c in remaining_for_spawning[:pick_count]:
                environment.assign_group(c, "spawn farmer")
            pick_index = pick_count
            wheat -= s_f * 10

        # Spawn warriors: need 2 villagers and 12 wheat per spawn
        remaining_after_farm_spawns = remaining_for_spawning[pick_index:] if pick_index > 0 else remaining_for_spawning
        max_possible_war_spawns = wheat // 12
        s_w = min(len(remaining_after_farm_spawns) // 2, max_possible_war_spawns)

        if s_w > 0:
            pick_count_w = 2 * s_w
            for c in remaining_after_farm_spawns[:pick_count_w]:
                environment.assign_group(c, "spawn warrior")
            wheat -= s_w * 12

        # Farmers not assigned to spawn groups stay in "farm" (no further action)
        # Warriors were already assigned to "cave" above; they will attack in the cave step.

    def assign_in_cave(self, components, environment, group_ids, step: int):
        """
        Divide Villagers in the Cave into:
        - "attack": Attack the Dragon
        - "cave": Stay in the Cave
        - "village": Go to the Village
        """
        # Warriors should attack; Farmers should go to village
        for c in components:
            if getattr(c, "role", None) == "Warrior":
                environment.assign_group(c, "attack")
            else:
                # Farmers go to village
                environment.assign_group(c, "village")
