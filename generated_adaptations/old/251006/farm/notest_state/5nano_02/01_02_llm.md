# Strategy reasoning:
# Objective: Allocate drones to protect fields from birds with a simple, robust rule:
# - Identify the field with the highest bird threat (threat_level > 0).
# - Fully defend that field using as many drones as needed to achieve full protection
#   (as defined by field.drones_for_full_protection). Use the closest available drones to
#   minimize travel time. If some drones are already protecting that field, keep them there.
# - Any drones not required for the top-field protection should be assigned to idle.
# - If there are no threatened fields, keep all drones idle.
# Rationale: This policy directly targets the most urgent risk and minimizes wasted movement
# by prioritizing proximity. It also ensures full protection for the most dangerous field, which
# aligns with the objective of minimizing overall damage.

import math
from generated_adaptations.base_classes.farm import FarmAdaptation


class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        # Build list of fields currently under threat (threat_level > 0)
        threatened_fields = []
        for field in environment.fields:
            lv = getattr(field, "threat_level", 0)
            if lv is None:
                lv = 0
            if lv > 0:
                threatened_fields.append(field)

        # If there are no threats, idle all drones
        if not threatened_fields:
            for d in components:
                environment.assign_group(d, "idle")
            return

        # Choose the top-threat field (highest threat_level)
        target_field = max(threatened_fields, key=lambda f: getattr(f, "threat_level", 0))

        # Group name for this field
        target_group = f"protecting {target_field.id}"
        if target_group not in group_ids:
            # If the group is not valid, fallback to idle
            for d in components:
                environment.assign_group(d, "idle")
            return

        # How many drones are needed to fully protect this field
        required = getattr(target_field, "drones_for_full_protection", 0)

        # Keep track of drones we assign to the top-field protection in this step
        protected_now = set()

        # First, re-assign any drones currently protecting this field to the target group
        for d in components:
            if getattr(d, "state", None) == "protecting" and getattr(d, "target_id", None) == target_field.id:
                environment.assign_group(d, target_group)
                protected_now.add(d)

        current_count = len(protected_now)

        # If more drones are needed to reach full protection, pick the closest available drones
        need = max(0, int(required) - current_count)
        if need > 0:
            # Candidates: drones not already protecting the target field
            candidates = [d for d in components if d not in protected_now]
            cx = (getattr(target_field, "left", 0) + getattr(target_field, "right", 0)) / 2.0
            cy = (getattr(target_field, "top", 0) + getattr(target_field, "bottom", 0)) / 2.0

            def dist_to_center(drone):
                loc = getattr(drone, "location", None)
                if loc is None:
                    return float("inf")
                return math.hypot(loc.x - cx, loc.y - cy)

            candidates.sort(key=dist_to_center)
            for i in range(min(need, len(candidates))):
                d = candidates[i]
                environment.assign_group(d, target_group)
                protected_now.add(d)

        # Finally, assign all drones not in the protected_now set to idle
        for d in components:
            if d in protected_now:
                continue
            environment.assign_group(d, "idle")
