# Reasoning and adaptation strategy:
# - Goal: Improve protection by aggressively but stably allocating drones to the most-threatened fields while minimizing unnecessary movement.
# - Key ideas:
#   1) Always attempt to fully protect the highest-threat field first, using the closest available drones.
#   2) If there are spare drones after top-field protection is satisfied, pre-position some of them to the second-most-threatened field (if any) to reduce future movement.
#   3) Preserve drones already protecting a field whenever possible, and only reassign when needed to reach a full protection target.
#   4) Ensure every drone is assigned to exactly one group (idle or protecting a specific field). Guard against invalid group names.
#
# This approach reduces average damage by keeping protection stable on the top field and strategically pre-positioning spares for the next-threat field, while avoiding unnecessary churn.

from generated_adaptations.base_classes.farm import FarmAdaptation

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        # Gather fields with positive threat
        fields = getattr(environment, "fields", [])
        threat_fields = [f for f in fields if getattr(f, "threat_level", 0) > 0]

        # If no threat, idle all drones
        if not threat_fields:
            for d in components:
                environment.assign_group(d, "idle")
            return

        # Sort threat fields by descending threat level
        threat_fields_sorted = sorted(
            threat_fields,
            key=lambda f: getattr(f, "threat_level", 0),
            reverse=True
        )

        # Helper to form group name
        def field_group(field):
            return f"protecting {field.id}"

        # Top field (highest threat)
        top = threat_fields_sorted[0]
        top_group = field_group(top)
        if top_group not in group_ids:
            # If the group name isn't valid, fall back to idle to avoid errors
            for d in components:
                environment.assign_group(d, "idle")
            return

        top_needed = int(getattr(top, "drones_for_full_protection", 0))

        # Current protectors for the top field
        current_top = [
            d for d in components
            if getattr(d, "state", None) == "protecting" and getattr(d, "target_id", None) == getattr(top, "id", None)
        ]
        current_top_set = set(current_top)

        final_top_protectors = set(current_top_set)

        # If we have meaningful protection requirement and not yet full, add closest drones
        if top_needed > 0 and len(current_top) < top_needed:
            remaining = top_needed - len(current_top)
            cx = (getattr(top, "left", 0) + getattr(top, "right", 0)) / 2.0
            cy = (getattr(top, "top", 0) + getattr(top, "bottom", 0)) / 2.0

            candidates = []
            for d in components:
                if d in current_top_set:
                    continue
                loc = getattr(d, "location", None)
                if loc is None:
                    dist = float("inf")
                else:
                    dx = getattr(loc, "x", 0)
                    dy = getattr(loc, "y", 0)
                    dist = ((dx - cx) ** 2 + (dy - cy) ** 2) ** 0.5
                candidates.append((dist, d))

            candidates.sort(key=lambda t: t[0])
            for _, d in candidates[:max(0, remaining)]:
                final_top_protectors.add(d)

        # Prepare second field (pre-positioning) if beneficial
        final_second_protectors = set()
        second_group = None

        if len(threat_fields_sorted) > 1:
            second = threat_fields_sorted[1]
            second_group_candidate = field_group(second)
            if second_group_candidate in group_ids:
                second_group = second_group_candidate
                second_needed = int(getattr(second, "drones_for_full_protection", 0))
                current_second = [
                    d for d in components
                    if getattr(d, "state", None) == "protecting" and getattr(d, "target_id", None) == getattr(second, "id", None)
                ]
                final_second_protectors = set(current_second)

                if second_needed > 0 and len(current_second) < second_needed:
                    remaining2 = second_needed - len(current_second)
                    cx2 = (getattr(second, "left", 0) + getattr(second, "right", 0)) / 2.0
                    cy2 = (getattr(second, "top", 0) + getattr(second, "bottom", 0)) / 2.0

                    # Drones available to be assigned to second field (not already committed to top field)
                    spare_for_second = [
                        d for d in components
                        if d not in final_top_protectors and d not in final_second_protectors
                    ]

                    cand2 = []
                    for d in spare_for_second:
                        loc = getattr(d, "location", None)
                        if loc is None:
                            dist = float("inf")
                        else:
                            dx = getattr(loc, "x", 0)
                            dy = getattr(loc, "y", 0)
                            dist = ((dx - cx2) ** 2 + (dy - cy2) ** 2) ** 0.5
                        cand2.append((dist, d))

                    cand2.sort(key=lambda t: t[0])
                    for _, d in cand2[:max(0, remaining2)]:
                        final_second_protectors.add(d)
            else:
                second_group = None  # invalid group name for second field

        # Final assignment: each drone goes to either top_protectors, second_protectors, or idle
        for d in components:
            if d in final_top_protectors:
                environment.assign_group(d, top_group)
            elif second_group is not None and d in final_second_protectors:
                environment.assign_group(d, second_group)
            else:
                environment.assign_group(d, "idle")
