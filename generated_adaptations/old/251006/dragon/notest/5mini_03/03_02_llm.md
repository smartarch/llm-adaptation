# SmartAdaptation v3 for the Dragon hunt problem.
#
# Reasoning and analysis (in comments):
#
# Observations from prior runs:
# - The Dragon retaliates when attacked; AoE (40%) and eat (20%) hits everyone in the Cave.
#   Therefore, minimizing the number of villagers in the Cave at the moment of attack reduces casualties.
# - Farmers produce wheat and are needed to spawn new villagers. Losing farmers or sending too many
#   of them to spawn at once harms long-term wheat income.
# - Spawning warrior pairs increases DPS but consumes wheat and requires sacrificing two villagers
#   temporarily to the spawn group.
#
# Goals for improvement (reduce damage to the fields):
# - Keep more farmers farming (increase average number of farming villagers).
# - Keep fewer villagers in the Cave simultaneously (send very small attack squads, ideally 1).
# - Build up farmer population early (spawn farmer pairs until a small target) so wheat income is steady.
# - Spawn warriors eventually, but avoid spawning too many pairs per step; prefer one pair when wheat allows.
# - Rotate which warrior attacks to distribute risk (so the same warrior is not always exposed).
#
# Strategy summary:
# 1) Attack squad: send exactly one warrior from the Village to the Cave each step (if any). Use rotation
#    so different warriors take turns attacking.
# 2) Farmers: maintain a target number of farmers (DESIRED_FARMERS). If current farmers < target and wheat >= 10,
#    spawn farmer pairs (one pair per step) until the target is reached or wheat exhausted. Always leave at least
#    one farmer farming.
# 3) Once farmer target met, prefer spawning a warrior pair when wheat >= 12 (one pair per step) to boost DPS.
# 4) Assign non-squad warriors to stay in Village (safe) by assigning them to "farm" group (Village group).
# 5) In Cave: allow at most one warrior to attack; send other warriors back to Village. Send all farmers back to Village.
# 6) HP-aware: do not send warriors with hp <= 1 to attack; send them back to Village to preserve them.
#
# Implementation notes:
# - The class keeps a small persistent state (next_attacker_index) to rotate attackers.
# - All components are explicitly assigned to exactly one group every call.
# - Code is defensive about available group names in group_ids and uses fallbacks.
#
# This tradeoff increases farming presence while keeping the attack exposure minimal, reducing field damage
# while still achieving consistent DPS over multiple steps.

from generated_adaptations.base_classes.dragon import DragonHuntAdaptation

class SmartAdaptation(DragonHuntAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        # Rotate which warrior is sent to attack (index into the current warriors list)
        self.next_attacker_index = 0
        # Configuration
        self.ATTACK_SQUAD_SIZE = 1        # send only one warrior to attack per step
        self.DESIRED_FARMERS = 4         # aim to grow farmers to this number for stable wheat production
        self.MAX_WARRIOR_PAIRS_PER_STEP = 1
        self.MAX_FARMER_PAIRS_PER_STEP = 1

    def assign_in_village(self, components, environment, group_ids, step: int):
        # Validate groups
        has_farm = "farm" in group_ids
        has_cave = "cave" in group_ids
        has_spawn_farmer = "spawn farmer" in group_ids
        has_spawn_warrior = "spawn warrior" in group_ids

        # Partition villagers by role
        farmers = [c for c in components if getattr(c, "role", "").lower() == "farmer"]
        warriors = [c for c in components if getattr(c, "role", "").lower() == "warrior"]

        # Read wheat
        try:
            wheat = int(getattr(environment.farm, "wheat", 0))
        except Exception:
            wheat = 0

        # Determine attack squad from warriors using rotation
        squad = []
        non_squad = []
        if len(warriors) > 0:
            # normalize next_attacker_index
            self.next_attacker_index = self.next_attacker_index % len(warriors) if len(warriors) > 0 else 0
            # Build squad by selecting up to ATTACK_SQUAD_SIZE healthy warriors (hp > 1)
            i = self.next_attacker_index
            selected = 0
            scanned = 0
            # Try to find up to ATTACK_SQUAD_SIZE warriors with hp>1 by scanning list cyclically
            while selected < self.ATTACK_SQUAD_SIZE and scanned < len(warriors):
                w = warriors[i % len(warriors)]
                if getattr(w, "hp", 0) > 1:
                    squad.append(w)
                    selected += 1
                i += 1
                scanned += 1
            # set next_attacker_index for rotation: move to next warrior (so rotation advances)
            self.next_attacker_index = (self.next_attacker_index + 1) % len(warriors)
            # Non-squad are those warriors not in squad
            non_squad = [w for w in warriors if w not in squad]
        else:
            squad = []
            non_squad = []

        # Assign squad to go to cave (they will attack there)
        for w in squad:
            target = "cave" if has_cave else ("farm" if has_farm else group_ids[0] if group_ids else "cave")
            environment.assign_group(w, target)

        # Assign non-squad warriors to stay in Village (use "farm" as a safe village group)
        for w in non_squad:
            target = "farm" if has_farm else (group_ids[0] if group_ids else "farm")
            environment.assign_group(w, target)

        # FARMER SPAWNING LOGIC
        # We will try to reach DESIRED_FARMERS by spawning farmer pairs (one pair per step) if wheat allows.
        # Otherwise, if target reached, we may spawn a warrior pair (one pair per step) if wheat >= 12.
        assigned_farmers = set()
        farmers_list = list(farmers)

        # Reserve at least one farmer to farm if possible (to keep production)
        reserve_min = 1
        # If there are only exactly 2 farmers and we are below desired, allow using both to spawn to grow quickly
        # (this is an aggressive early-growth rule)
        if len(farmers_list) <= 2 and len(farmers_list) < self.DESIRED_FARMERS and wheat >= 10 and has_spawn_farmer:
            # Spawn one farmer pair using both farmers
            if len(farmers_list) == 2:
                f1, f2 = farmers_list[0], farmers_list[1]
                environment.assign_group(f1, "spawn farmer" if has_spawn_farmer else ("farm" if has_farm else group_ids[0]))
                environment.assign_group(f2, "spawn farmer" if has_spawn_farmer else ("farm" if has_farm else group_ids[0]))
                assigned_farmers.update({f1, f2})
                wheat -= 10
        else:
            # If we still have fewer farmers than desired, try to spawn one farmer pair (if possible)
            if len(farmers_list) < self.DESIRED_FARMERS and wheat >= 10 and has_spawn_farmer:
                # ensure we leave at least one farmer farming if possible
                available_for_spawn = max(0, len(farmers_list) - reserve_min)
                if available_for_spawn >= 2:
                    # assign first two available to spawn farmer
                    f1 = farmers_list[0]
                    f2 = farmers_list[1]
                    environment.assign_group(f1, "spawn farmer" if has_spawn_farmer else ("farm" if has_farm else group_ids[0]))
                    environment.assign_group(f2, "spawn farmer" if has_spawn_farmer else ("farm" if has_farm else group_ids[0]))
                    assigned_farmers.update({f1, f2})
                    wheat -= 10
            else:
                # If farmer target met (or spawning farmers not desired), consider spawning a warrior pair (one pair per step)
                # Only attempt if we have at least 2 available farmers (leave reserve_min)
                available_for_spawn = max(0, len(farmers_list) - reserve_min)
                if wheat >= 12 and available_for_spawn >= 2 and has_spawn_warrior:
                    # spawn one warrior pair
                    f1 = farmers_list[0]
                    # pick next available farmer that isn't the same
                    # find second farmer index
                    idx2 = 1 if len(farmers_list) > 1 else None
                    if idx2 is not None:
                        f2 = farmers_list[idx2]
                        environment.assign_group(f1, "spawn warrior")
                        environment.assign_group(f2, "spawn warrior")
                        assigned_farmers.update({f1, f2})
                        wheat -= 12

        # Assign any farmers not already assigned to farming
        for f in farmers_list:
            if f in assigned_farmers:
                continue
            target = "farm" if has_farm else (group_ids[0] if group_ids else "farm")
            environment.assign_group(f, target)

    def assign_in_cave(self, components, environment, group_ids, step: int):
        # Validate groups
        has_attack = "attack" in group_ids
        has_village = "village" in group_ids
        has_cave = "cave" in group_ids

        # Identify warriors in cave and limit number of attackers to ATTACK_SQUAD_SIZE
        warriors_in_cave = [c for c in components if getattr(c, "role", "").lower() == "warrior"]
        farmers_in_cave = [c for c in components if getattr(c, "role", "").lower() == "farmer"]

        # Decide which warriors attack: prefer healthy ones (hp > 1), up to ATTACK_SQUAD_SIZE.
        attackers = []
        others = []
        healthy = [w for w in warriors_in_cave if getattr(w, "hp", 0) > 1]
        weak = [w for w in warriors_in_cave if getattr(w, "hp", 0) <= 1]
        # select up to ATTACK_SQUAD_SIZE healthy attackers
        attackers = healthy[:self.ATTACK_SQUAD_SIZE]
        # others (including weak and surplus healthy) should be sent back to village to avoid AoE deaths
        others = [w for w in warriors_in_cave if w not in attackers]

        # Assign attackers to "attack"
        for a in attackers:
            target = "attack" if has_attack else ("cave" if has_cave else (group_ids[0] if group_ids else "attack"))
            environment.assign_group(a, target)

        # Send other warriors back to village (safe)
        for w in others:
            target = "village" if has_village else (group_ids[0] if group_ids else "village")
            environment.assign_group(w, target)

        # Send all farmers in cave back to village to continue farming/spawning
        for f in farmers_in_cave:
            target = "village" if has_village else (group_ids[0] if group_ids else "village")
            environment.assign_group(f, target)
