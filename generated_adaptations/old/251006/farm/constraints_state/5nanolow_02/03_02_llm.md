import abc
import math

try:
    from generated_adaptations.base_classes.farm import FarmAdaptation
except Exception:
    class FarmAdaptation(abc.ABC):
        def __init__(self, **kwargs):
            super().__init__()

        @abc.abstractmethod
        def assign_drones(self, components, environment, group_ids, step: int):
            pass


class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def _field_center(self, field):
        return ((field.left + field.right) / 2.0, (field.top + field.bottom) / 2.0)

    def _dist(self, p, q):
        dx = p[0] - q[0]
        dy = p[1] - q[1]
        return math.hypot(dx, dy)

    def assign_drones(self, components, environment, group_ids, step: int):
        # 1) Identify fields with threat > 0 and pick the most threatened
        fields = [f for f in environment.fields if f.threat_level > 0]
        if not fields:
            for c in components:
                environment.assign_group(c, "idle")
            return

        fields.sort(key=lambda f: f.threat_level, reverse=True)
        top_field = fields[0]
        top_group = f"protecting {top_field.id}"
        if top_group not in group_ids:
            top_group = "idle"

        # 2) Prepare center for distance calculations
        center_top = self._field_center(top_field)

        # 3) Start with drones already protecting or en route to top_field
        planned = set()
        currently = [d for d in components if (d.state == "protecting" and d.target_id == top_field.id) or
                                     (d.state == "moving_to_field" and d.target_id == top_field.id)]
        for d in currently:
            planned.add(id(d))

        # 4) Fill to drones_for_full_protection with nearest available drones
        needed = getattr(top_field, "drones_for_full_protection", 0)
        current_count = len(currently)

        # Assign currently protecting/moving drones to the top group (idempotent by plan)
        for d in currently:
            environment.assign_group(d, top_group)

        if current_count < needed:
            candidates = []
            for d in components:
                if id(d) in planned:
                    continue
                pos = d.location
                dist = self._dist((pos.x, pos.y), center_top)
                candidates.append((dist, d))
            candidates.sort(key=lambda t: t[0])
            to_take = min(needed - current_count, len(candidates))
            for i in range(to_take):
                d = candidates[i][1]
                planned.add(id(d))
                environment.assign_group(d, top_group)

        # 5) All other drones stay idle to minimize movement
        for d in components:
            if id(d) not in planned and id(d) not in {id(x) for x in currently}:
                environment.assign_group(d, "idle")

        # Note: We deliberately do not assign drones to secondary fields to keep movement low
        # and to maximize protection of the top-threat field per the guidance.
