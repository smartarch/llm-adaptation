Reasoning and updated adaptation strategy:
- Goal: further reduce the risk to fields while maintaining or increasing Dragon DPS, aiming for fast, reliable wins.
- Core adjustments:
  - Minimize cave exposure: limit the number of villagers in the Cave at any given time. Specifically, keep at most one Warrior in the cave per village turn, and rotate attackers so DPS scales up over time without creating large AoE casualties.
  - Controlled wave spawning: spawn Warriors only when there are enough Farmers and Wheat. Use pairs of Farmers to spawn Warriors (12 wheat per Warrior) and pairs of Farmers to spawn Farmers (10 wheat per new Farmer). This creates a gradual but steady growth of both DPS and wheat production.
  - Farmers stay in Village: always prioritize farming to sustain future spawns, unless a spawn condition explicitly uses farmers for spawning.
  - In the Cave phase, all Warriors in the cave attack, while Farmers retreat to the Village. This minimizes the number of villagers exposed to Dragon AoE when the Dragon responds.
- Expected outcome: fewer deaths due to AoE in the Cave, while gradually ramping up waves of Warriors to increase damage and still hitting the 30-turn win condition.

from generated_adaptations.base_classes.dragon import DragonHuntAdaptation

class SmartAdaptation(DragonHuntAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_in_village(self, components, environment, group_ids, step: int):
        # Split villagers by role
        farmers = [c for c in components if c.role == "Farmer"]
        warriors = [c for c in components if c.role == "Warrior"]
        assigned = set()

        # Current wheat in farm
        wheat = getattr(environment.farm, "wheat", 0)

        # 1) Spawn Warriors if possible: for every 2 farmers and 12 wheat -> 1 Warrior
        max_war_spawns = min(len(farmers) // 2, wheat // 12)
        if max_war_spawns > 0:
            to_use = farmers[:2 * max_war_spawns]
            for i in range(max_war_spawns):
                a = to_use[2 * i]
                b = to_use[2 * i + 1]
                environment.assign_group(a, "spawn warrior")
                environment.assign_group(b, "spawn warrior")
                assigned.add(a)
                assigned.add(b)

        # 2) Spawn Farmers if possible: for every 2 remaining farmers and 10 wheat -> 1 Farmer
        remaining_farmers = [f for f in farmers if f not in assigned]
        max_farm_spawns = min(len(remaining_farmers) // 2, wheat // 10)
        if max_farm_spawns > 0:
            to_use_f = remaining_farmers[:2 * max_farm_spawns]
            for i in range(max_farm_spawns):
                a = to_use_f[2 * i]
                b = to_use_f[2 * i + 1]
                environment.assign_group(a, "spawn farmer")
                environment.assign_group(b, "spawn farmer")
                assigned.add(a)
                assigned.add(b)

        # 3) After spawns, assign remaining farmers to farming
        remaining_farmers = [f for f in farmers if f not in assigned]

        # 4) Maintain exactly one Warrior in cave (if available) to minimize AoE risk
        cave_target = None
        free_warriors = [w for w in warriors if w not in assigned]
        if len(free_warriors) > 0:
            cave_target = free_warriors[0]
            environment.assign_group(cave_target, "cave")
            assigned.add(cave_target)

        # 5) All other villagers go to farming by default
        for f in remaining_farmers:
            if f in assigned:
                continue
            environment.assign_group(f, "farm")
            assigned.add(f)

        # 6) Any remaining warriors (beyond the one sent to cave) go to farming
        for w in free_warriors:
            if w in assigned:
                continue
            environment.assign_group(w, "farm")
            assigned.add(w)

    def assign_in_cave(self, components, environment, group_ids, step: int):
        # In the cave: all Warriors attack; Farmers go back to the village
        for c in components:
            if c.role == "Warrior":
                environment.assign_group(c, "attack")
            else:
                environment.assign_group(c, "village")
