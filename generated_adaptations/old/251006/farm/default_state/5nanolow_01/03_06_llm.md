from typing import List
import math

# Import the base class from the expected location
from generated_adaptations.base_classes.farm import FarmAdaptation

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
    
    def _field_center(self, field) -> tuple:
        cx = (field.left + field.right) / 2.0
        cy = (field.top + field.bottom) / 2.0
        return (cx, cy)
    
    def _distance(self, p, q) -> float:
        dx = p.x - q[0]
        dy = p.y - q[1]
        return math.hypot(dx, dy)
    
    def assign_drones(self, components: List, environment, group_ids: List[str], step: int):
        valid_groups = set(group_ids)
        # Gather fields with threat > 0, sorted by threat descending
        fields_with_threat = [f for f in environment.fields if getattr(f, "threat_level", 0) > 0]
        if not fields_with_threat:
            # Assign everyone to idle (explicitly)
            for comp in components:
                environment.assign_group(comp, "idle" if "idle" in valid_groups else "idle")
            return
        
        fields_with_threat.sort(key=lambda f: f.threat_level, reverse=True)
        
        total_drones = len(components)
        
        # Determine current protecting per field
        current_protecting_by_field = {}
        for field in fields_with_threat:
            current = 0
            for comp in components:
                if getattr(comp, "state", None) == "protecting" and getattr(comp, "target_id", None) == field.id:
                    current += 1
            current_protecting_by_field[field] = current
        
        # Determine kept drones for full protection and needs per field
        kept_drones = set()
        field_needs = {}
        for field in fields_with_threat:
            kept_for_field = min(current_protecting_by_field.get(field, 0), getattr(field, "drones_for_full_protection", 0))
            # Mark kept drones (any protecting towards this field) up to kept_for_field
            kept = 0
            for comp in components:
                if kept >= kept_for_field:
                    break
                if getattr(comp, "state", None) == "protecting" and getattr(comp, "target_id", None) == field.id:
                    kept_drones.add(comp)
                    kept += 1
            need = int(getattr(field, "drones_for_full_protection", 0) - kept_for_field)
            if need > 0:
                field_needs[field] = need
        
        # Prepare final assignment map (component -> group)
        comp_to_group = {comp: "idle" for comp in components}
        
        # Reassign kept drones to their current protecting targets
        for comp in kept_drones:
            if getattr(comp, "state", None) == "protecting":
                comp_to_group[comp] = f"protecting {getattr(comp, 'target_id', None)}"
            else:
                comp_to_group[comp] = "idle"
        
        # If nothing needs protection, keep kept ones and leave others idle
        if not field_needs:
            for comp in components:
                environment.assign_group(comp, comp_to_group[comp])
            return
        
        # Pool of available drones to allocate (exclude kept drones)
        available = [c for c in components if c not in kept_drones]
        
        # Allocate across fields in threat order
        for field in fields_with_threat:
            if field not in field_needs:
                continue
            need = field_needs[field]
            if need <= 0:
                continue
            center = self._field_center(field)
            cand = []
            for c in available:
                pos = getattr(c, "location", None)
                dist = float('inf')
                if pos is not None:
                    dist = self._distance(pos, center)
                cand.append((dist, c))
            cand.sort(key=lambda x: x[0])
            take = min(need, len(cand))
            for i in range(take):
                _, c = cand[i]
                comp_to_group[c] = f"protecting {field.id}"
                available.remove(c)
        
        # After allocation, set final groups, ensuring every drone is assigned exactly once
        for comp in components:
            environment.assign_group(comp, comp_to_group[comp])
