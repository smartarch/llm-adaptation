# Strategy: per-drone greedy assignment to avoid multiple passes and ensure
# the most threatened fields receive protection first. Drones already protecting
# a field stay assigned to that field. Remaining drones are allocated one by one
# to the highest-threat field that still needs protection, until all needs are met.

from typing import List

from generated_adaptations.base_classes.farm import FarmAdaptation  # type: ignore

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components: List, environment, group_ids: List[str], step: int):
        # 1) Gather fields with positive threat and sort by threat (desc)
        fields_with_threat = [
            f for f in environment.fields if getattr(f, "threat_level", 0.0) > 0.0
        ]
        fields_with_threat.sort(
            key=lambda f: float(getattr(f, "threat_level", 0.0)), reverse=True
        )

        if not fields_with_threat:
            # No threat: idle all drones
            for c in components:
                environment.assign_group(c, "idle")
            return

        # 2) Compute how many drones are currently protecting each field
        currently_protecting = {
            f.id: 0 for f in fields_with_threat
        }
        # Also track if drones are already protecting a field (to keep them)
        # We'll use a quick lookup: for any drone with state == "protecting", count it for its target
        for c in components:
            if getattr(c, "state", None) == "protecting":
                tid = getattr(c, "target_id", None)
                if tid is not None:
                    currently_protecting[tid] = currently_protecting.get(tid, 0) + 1

        # 3) Compute remaining needs for each field (only fields with threat > 0)
        needs = {}
        for f in fields_with_threat:
            field_id = f.id
            required = 0
            try:
                required = int(getattr(f, "drones_for_full_protection", 0))
            except Exception:
                required = 0
            current = currently_protecting.get(field_id, 0)
            needs[field_id] = max(0, required - current)

        # 4) Assign drones one by one
        for c in components:
            state = getattr(c, "state", None)
            if state == "protecting":
                # Keep current protection group for drones already protecting
                field_id = getattr(c, "target_id", None)
                if field_id is not None:
                    environment.assign_group(c, f"protecting {field_id}")
                else:
                    environment.assign_group(c, "idle")
                continue

            # If there is any field that still needs protection, assign to the highest-threat one
            target_field_id = None
            for f in fields_with_threat:
                fid = f.id
                if needs.get(fid, 0) > 0:
                    target_field_id = fid
                    needs[fid] -= 1  # this drone will contribute to this field
                    break

            if target_field_id is not None:
                environment.assign_group(c, f"protecting {target_field_id}")
            else:
                environment.assign_group(c, "idle")
