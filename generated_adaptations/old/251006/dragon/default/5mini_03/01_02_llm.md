"""
SmartAdaptation for Dragon hunting.

Reasoning and step-by-step analysis:
- Objective: kill the Dragon as fast as possible. Warriors are the main damage dealers (3 dmg each),
  Farmers produce wheat (to pay for spawns) and provide bodies for spawning new villagers.
- Constraints:
  - All Warriors should go to the Cave and attack the Dragon.
  - All Farmers should remain in the Village to farm or be assigned to spawn groups (spawn farmer / spawn warrior).
  - Spawning consumes wheat (10 for farmer, 12 for warrior) and requires pairs of villagers assigned to the appropriate spawn group.
  - Keep at least some farmers farming so wheat production continues; otherwise spawns will starve future production.
  - If farmers end up in the Cave, they should be sent back to the Village.

Adaptation strategy (high level):
1. assign_in_cave:
   - Warriors: assign to "attack".
   - Farmers: send back to "village".
   Rationale: Warriors are the only ones who should fight in the Cave. Farmers do not contribute effectively to fighting,
   and are needed to produce wheat and be used for spawning.

2. assign_in_village:
   - Immediately send any Warriors in the Village to the Cave ("cave" group).
   - For Farmers:
     - Maintain at least one farmer actively farming to ensure ongoing wheat production.
     - Use remaining farmers in pairs to spawn new villagers when wheat allows.
     - Prioritize spawning Warriors when the Dragon still has significant HP (we need damage output)
       or when enough wheat is available to afford Warriors; otherwise spawn Farmers to increase wheat
       production capacity and future spawning throughput.
     - Assign farmers participating in spawning to "spawn warrior" or "spawn farmer" in pairs so the environment
       can consume wheat and create new villagers.
     - Any farmers not used for spawning remain assigned to "farm".
   Rationale:
     - This balances immediate damage (spawn warriors when needed) and sustained economy (spawn farmers when we
       need more wheat production or cannot afford warriors).
     - Keeping one farmer farming prevents total starvation of wheat income.

Implementation notes:
- All assignments are done via environment.assign_group(component, group_id).
- The code respects the requirement that each component must be explicitly assigned to exactly one group.
- The strategy is conservative about removing too many farmers from farming: it keeps min_farmers_farming = 1.
- The spawn allocation loop consumes the available farmers in pairs and the observed wheat amount
  to choose spawn type step by step until resources or available farmers are exhausted.

"""

from generated_adaptations.base_classes.dragon import DragonHuntAdaptation


class SmartAdaptation(DragonHuntAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_in_village(self, components, environment, group_ids, step: int):
        """
        components: list of villagers currently in Village
        environment.farm.wheat: current wheat
        environment.dragon.hp: dragon hp
        Groups available: "farm", "cave", "spawn farmer", "spawn warrior"
        """
        # Ensure group names are used exactly as specified
        GROUP_FARM = "farm"
        GROUP_CAVE = "cave"
        GROUP_SPAWN_FARMER = "spawn farmer"
        GROUP_SPAWN_WARRIOR = "spawn warrior"

        # Partition components by role
        farmers = [c for c in components if getattr(c, "role", "") == "Farmer"]
        warriors = [c for c in components if getattr(c, "role", "") == "Warrior"]

        # 1) Send all warriors to the Cave
        for w in warriors:
            environment.assign_group(w, GROUP_CAVE)

        # 2) For farmers, decide who farms and who goes to spawn groups
        wheat = 0
        try:
            wheat = int(environment.farm.wheat)
        except Exception:
            # if read fails for some reason, assume 0 to be safe
            wheat = 0

        dragon_hp = 0
        try:
            dragon_hp = int(environment.dragon.hp)
        except Exception:
            dragon_hp = 0

        # Keep at least this many farmers farming to ensure production
        min_farmers_farming = 1
        # Copy list of farmers to mutate
        remaining_farmers = list(farmers)

        # If there are fewer farmers than the minimum, just assign all to farm
        if len(remaining_farmers) <= min_farmers_farming:
            for f in remaining_farmers:
                environment.assign_group(f, GROUP_FARM)
            return

        # Spawn allocation loop:
        # We will iteratively assign pairs of farmers to spawn warrior or spawn farmer
        # Prioritize spawning warriors when Dragon HP is high (>25) or when we have ample wheat.
        # Otherwise spawn farmers to grow wheat income.
        assigned_to_spawn = []  # tuples (component, group)
        # Helper to pop two farmers for a spawn (if available)
        def pop_two_farmers():
            if len(remaining_farmers) >= 2:
                a = remaining_farmers.pop(0)
                b = remaining_farmers.pop(0)
                return [a, b]
            return None

        # While we have at least two farmers spare beyond the min farming and enough wheat, spawn
        while True:
            if len(remaining_farmers) <= min_farmers_farming:
                break  # keep minimum farmers farming
            # available pairs
            available_pairs = (len(remaining_farmers) - min_farmers_farming) // 2
            if available_pairs <= 0:
                break

            # Decide spawn type
            # Conditions to spawn warrior:
            # - Enough wheat for warrior (12)
            # - Prefer warriors if dragon still has significant HP or we have surplus wheat
            prefer_warrior = False
            if wheat >= 12 and (dragon_hp > 25 or wheat >= 24):
                prefer_warrior = True

            # Try to spawn warrior if preferred and possible
            if prefer_warrior and wheat >= 12:
                pair = pop_two_farmers()
                if pair is None:
                    break
                for f in pair:
                    assigned_to_spawn.append((f, GROUP_SPAWN_WARRIOR))
                wheat -= 12
                continue

            # Otherwise try to spawn farmer if possible
            if wheat >= 10:
                pair = pop_two_farmers()
                if pair is None:
                    break
                for f in pair:
                    assigned_to_spawn.append((f, GROUP_SPAWN_FARMER))
                wheat -= 10
                continue

            # Can't spawn either due to lack of wheat
            break

        # Assign spawn participants
        for comp, grp in assigned_to_spawn:
            environment.assign_group(comp, grp)

        # Any remaining farmers (including the reserved min_farmers_farming) should farm
        for f in remaining_farmers:
            environment.assign_group(f, GROUP_FARM)

    def assign_in_cave(self, components, environment, group_ids, step: int):
        """
        components: list of villagers currently in Cave
        Groups available: "attack", "cave", "village"
        Behavior:
          - Warriors => "attack"
          - Farmers => "village" (send back to Village)
        """
        GROUP_ATTACK = "attack"
        GROUP_CAVE = "cave"
        GROUP_VILLAGE = "village"

        for c in components:
            role = getattr(c, "role", "")
            if role == "Warrior":
                environment.assign_group(c, GROUP_ATTACK)
            else:
                # Farmers should be returned to Village to farm / spawn
                environment.assign_group(c, GROUP_VILLAGE)
