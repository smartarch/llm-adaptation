import abc

# The base class is expected to be importable from the given path
from generated_adaptations.base_classes.farm import FarmAdaptation as _BaseFarmAdaptation

class SmartFarmAdaptation(_BaseFarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_drones(self, components, environment, group_ids, step: int):
        """
        Improved adaptation strategy:
        - Identify all fields with threat_level > 0 and sort them by threat (high -> low).
        - Aim to maximize protection coverage by filling fields in that order, up to full protection.
        - Preserve drones already protecting a field unless moving to a higher-threat field is necessary.
        - When allocating, prefer drones that are idle or moving to a field (to minimize disruption),
          before reallocating drones currently protecting other fields.
        - Each drone is assigned exactly once in this call.
        """
        # Gather fields
        fields = list(environment.fields)

        # Current protection per field (count drones currently protecting that field)
        current_protecting = {field.id: 0 for field in fields}
        for d in components:
            if d.state == "protecting" and d.target_id is not None:
                current_protecting[d.target_id] = current_protecting.get(d.target_id, 0) + 1

        # Candidate fields with threat > 0, sorted by threat high -> low
        candidates = [f for f in fields if getattr(f, "threat_level", 0) > 0]
        candidates.sort(key=lambda ff: ff.threat_level, reverse=True)

        # If no field requires attention, idle all
        if not candidates:
            for d in components:
                environment.assign_group(d, "idle")
            return

        # We will compute a final assignment map: drone -> group
        final_group = {}

        # Helper to set and avoid duplicates
        def assign(d, grp):
            final_group[d] = grp

        # First, try to fulfill full protection for each candidate field in order
        # Start with drones that are already protecting any field; keep them if their field is a candidate
        for f in candidates:
            field_id = f.id
            needed = max(0, getattr(f, "drones_for_full_protection", 0) - current_protecting.get(field_id, 0))

            if needed == 0:
                # Field already fully protected: ensure those drones stay protecting it
                for d in components:
                    if d.state == "protecting" and d.target_id == field_id:
                        assign(d, f"protecting {field_id}")
                # No more to do for this field
                continue

            # Ensure currently protecting drones for this field stay
            for d in components:
                if d.state == "protecting" and d.target_id == field_id:
                    assign(d, f"protecting {field_id}")

            # Allocate additional drones to this field, Prefer idle/moving_to_field first
            remaining = needed
            # Build a pool order: idle, moving_to_field, protecting other fields
            pool_order = sorted(components, key=lambda x: (
                x.state != "idle",                      # idle first (False -> 0), others 1
                x.state != "moving_to_field"            # moving_to_field second
            ))
            # Iterate pool and assign until remaining == 0
            for d in pool_order:
                if remaining <= 0:
                    break
                if final_group.get(d) is not None:
                    continue
                # Do not reassign drones that are already protecting this field (they are already handled)
                if d.state == "protecting" and d.target_id == field_id:
                    assign(d, f"protecting {field_id}")
                    remaining -= 1
                    continue
                # Assign this drone to protect the field
                assign(d, f"protecting {field_id}")
                remaining -= 1

        # Any drones not assigned yet: idle
        for d in components:
            if d not in final_group:
                assign(d, "idle")

        # Apply final assignments (exactly one call per drone)
        for d in components:
            environment.assign_group(d, final_group[d])
