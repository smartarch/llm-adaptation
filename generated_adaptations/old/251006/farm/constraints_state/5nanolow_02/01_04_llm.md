import abc
import math

# Assuming the base class is available from the given module path
try:
    from generated_adaptations.base_classes.farm import FarmAdaptation
except Exception:

class SmartFarmAdaptation(FarmAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def _field_center(self, field):
        cx = (field.left + field.right) / 2.0
        cy = (field.top + field.bottom) / 2.0
        return cx, cy

    def _distance(self, loc, point):
        dx = loc.x - point[0]
        dy = loc.y - point[1]
        return math.hypot(dx, dy)

    def assign_drones(self, components, environment, group_ids, step: int):
        # Identify fields with threat_level > 0
        candidate_fields = [f for f in environment.fields if f.threat_level > 0]

        # If no threat, idle all drones
        if not candidate_fields:
            for c in components:
                environment.assign_group(c, "idle")
            return

        # Prepare mapping for field groups
        field_groups = {f"protecting {f.id}": f for f in candidate_fields}

        # Compute centers for all candidate fields
        centers = {f.id: self._field_center(f) for f in candidate_fields}

        # Start with all drones unassigned
        assigned = set()

        # Helper to assign a drone to a group
        def assign_to_group(drone, group_name):
            environment.assign_group(drone, group_name)
            assigned.add(drone)

        # 1) Prioritize top field by threat level: ensure full protection
        top_field = max(candidate_fields, key=lambda f: f.threat_level)
        top_group = f"protecting {top_field.id}"
        if top_group not in group_ids:
            # Fallback to idle if group not defined
            top_group = "idle"

        # Current protecting drones for top field
        currently_protecting = [
            c for c in components if c.state == "protecting" and c.target_id == top_field.id
        ]
        current_count = len(currently_protecting)
        needed = getattr(top_field, "drones_for_full_protection", 0)

        # Protect top field up to full protection
        for c in currently_protecting:
            assign_to_group(c, top_group)
        for c in currently_protecting:
            pass  # already assigned

        # If not enough drones protecting top field, pick closest available
        if current_count < needed:
            # Build list of available drones not yet assigned
            available = []
            for c in components:
                if c in assigned or c in currently_protecting:
                    continue
                dist = self._distance(c.location, centers[top_field.id])
                available.append((dist, c))
            available.sort(key=lambda t: t[0])
            to_take = min(needed - current_count, len(available))
            for i in range(to_take):
                _, drone = available[i]
                assign_to_group(drone, top_group)

        # If more drones are needed beyond top field's full protection, or to utilize more drones,
        # distribute to other fields based on their threat level and proximity, up to their
        # drones_for_full_protection if provided.
        # Build a list of remaining fields (excluding top_field) with threat > 0
        remaining_fields = [f for f in candidate_fields if f.id != top_field.id]
        # Sort by threat level descending
        remaining_fields.sort(key=lambda f: f.threat_level, reverse=True)

        # For each remaining field, try to bring protection up to drones_for_full_protection
        for f in remaining_fields:
            group_name = f"protecting {f.id}"
            if group_name not in group_ids:
                continue
            # Current protecting drones for this field
            current = [c for c in components if c.state == "protecting" and c.target_id == f.id]
            curr_count = len(current)
            target_need = getattr(f, "drones_for_full_protection", 0)
            if curr_count >= target_need:
                # Already fully protected or at desired level
                for c in current:
                    assign_to_group(c, group_name)
                continue

            # Prepare list of available drones not yet assigned
            # Note: exclude those already assigned to any group
            available = []
            for c in components:
                if c in assigned:
                    continue
                dist = self._distance(c.location, self._field_center(f))
                available.append((dist, c))
            available.sort(key=lambda t: t[0])
            needed_more = min(target_need - curr_count, len(available))
            for i in range(needed_more):
                _, drone = available[i]
                assign_to_group(drone, group_name)

        # Any drone not assigned yet becomes idle
        for c in components:
            if c not in assigned:
                environment.assign_group(c, "idle")
