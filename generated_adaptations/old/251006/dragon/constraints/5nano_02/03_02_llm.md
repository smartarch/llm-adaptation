# Reasoning and adaptation strategy:
# Objective: Further reduce wheat (fields) damage while still achieving fast Dragon kill.
# Observations:
# - Warriors must go to the Cave to attack the Dragon; Farmers stay in the Village.
# - Spawning uses two villagers plus Wheat (12 for Warriors, 10 for Farmers) to create new villagers.
# - Wheat is produced by Farmers (5 wheats per farming action) and is consumed when spawning.
# - Dragon retaliation can kill villagers in the Cave; we want to minimize exposure by not flooding the Cave with villagers at once.
# Strategy:
# - Keep the rule "All Warriors go to the Cave" unchanged.
# - In the Village, implement a conservative, rhythm-based spawning policy:
#   - Use at most 1 Warrior spawn every 2 steps (when possible) to grow DPS gradually without large wheat spikes.
#   - Use at most 1 Farmer spawn every 2 steps (on alternate steps) if wheat and farmer pool allow.
#   - Always prioritize maintaining a healthy farming pool to sustain wheat production for future spawns.
# - This pacing reduces wheat depletion peaks, lowers risk of starving future spawns, and still achieves timely Dragon kill in practice.
# - Implement step-based gating using the provided step parameter to alternate between spawns.

from generated_adaptations.base_classes.dragon import DragonHuntAdaptation

class SmartAdaptation(DragonHuntAdaptation):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)

    def assign_in_village(self, components, environment, group_ids, step: int):
        # Separate villagers by role
        farmers = [c for c in components if getattr(c, "role", None) == "Farmer"]
        warriors = [c for c in components if getattr(c, "role", None) == "Warrior"]

        # Wheat available in the Farm (default to 0 if not present)
        available_wheat = 0
        try:
            available_wheat = int(getattr(environment.farm, "wheat", 0))
        except Exception:
            available_wheat = 0

        num_farmers = len(farmers)

        # Step-based spawning policy:
        # - Attempt at most 1 Warrior spawn every 2 steps (when possible)
        can_spawn_warrior = (step % 2 == 0) and (num_farmers >= 2) and (available_wheat >= 12)
        max_war_spawns = 1 if can_spawn_warrior else 0

        # Update hypothetical wheat and farmers after potential Warrior spawn
        remaining_wheat_after_war = available_wheat - max_war_spawns * 12
        remaining_farmers_after_war = num_farmers - max_war_spawns * 2

        # - Attempt at most 1 Farmer spawn every 2 steps (alternate step)
        can_spawn_farmer = (step % 2 == 1) and (remaining_farmers_after_war >= 2) and (remaining_wheat_after_war >= 10)
        max_farm_spawns = 1 if can_spawn_farmer else 0

        # Indices for selecting farmers to spawn
        # 1) Warrior spawn uses first 2 farmers
        spawn_warrior_vills = farmers[0 : 2 * max_war_spawns]

        # 2) Farmer spawn uses next 2 farmers after any Warrior spawn
        start_for_farm_spawns = 2 * max_war_spawns
        spawn_farmer_vills = farmers[start_for_farm_spawns : start_for_farm_spawns + 2 * max_farm_spawns]

        # 3) Remaining farmers go to farming
        remaining_farmers = farmers[start_for_farm_spawns + 2 * max_farm_spawns :]

        # Assign groups
        # 1) Warriors go to Cave (attack)
        for w in warriors:
            environment.assign_group(w, "cave")

        # 2) Spawn Warrior: two farmers per spawn
        for f in spawn_warrior_vills:
            environment.assign_group(f, "spawn warrior")

        # 3) Spawn Farmer: two farmers per spawn
        for f in spawn_farmer_vills:
            environment.assign_group(f, "spawn farmer")

        # 4) Remaining farmers go to farming in the Village
        for f in remaining_farmers:
            environment.assign_group(f, "farm")

    def assign_in_cave(self, components, environment, group_ids, step: int):
        # In the cave, all Warriors should attack the Dragon.
        # Farmers stay in the Village (not sent to cave this turn).
        for c in components:
            if getattr(c, "role", None) == "Warrior":
                environment.assign_group(c, "attack")
            else:
                environment.assign_group(c, "village")
